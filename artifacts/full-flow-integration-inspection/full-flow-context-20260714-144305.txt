import { createInterface } from 'node:readline/promises'
import { stdin, stdout } from 'node:process'
import { join } from 'node:path'
import { chromium, type Locator, type Page } from 'playwright'
import { PlatformPageManager } from '../src/automation/browser/platform-page-manager'
import { CpiQueueAvailabilityReader } from '../src/automation/cpi/cpi-queue-availability'
import { CpiQueueFilterController } from '../src/automation/cpi/cpi-queue-filter-controller'
import { CpiQueueMonitor } from '../src/automation/cpi/cpi-queue-monitor'

interface RawElement {
  tag: string
  id: string
  name: string
  type: string
  className: string
  title: string
  role: string
  href: string
  onclick: string
  target: string
  text: string
  value: string
}

function findCpiPage(pages: Page[]): Page | undefined {
  return pages.find((page) => page.url().includes('adm.cpicentral.com.br'))
}

function sanitizeUrl(value: string): string {
  if (!value) {
    return ''
  }

  try {
    const url = new URL(value)

    return `${url.origin}${url.pathname}`
  } catch {
    return value.split('?')[0].split('#')[0]
  }
}

function sanitizeText(value: string): string {
  const normalized = value.replace(/\s+/g, ' ').trim()

  if (!normalized) {
    return ''
  }

  const safeLabels = /^(abrir|alterar|consultar|detalhes|editar|responder|selecionar|visualizar)$/i

  if (safeLabels.test(normalized)) {
    return normalized
  }

  return `[conteúdo ocultado: ${normalized.length} caracteres]`
}

function sanitizeAttribute(value: string): string {
  const normalized = value.replace(/\s+/g, ' ').trim()

  if (!normalized) {
    return ''
  }

  if (normalized.includes('@') || /\d{3,}/.test(normalized)) {
    return '[ocultado]'
  }

  return normalized.slice(0, 120)
}

function sanitizeJavascript(value: string): string {
  const normalized = value.replace(/\s+/g, ' ').trim()

  if (!normalized) {
    return ''
  }

  return normalized
    .replace(/[\w.+-]+@[\w.-]+\.[A-Za-z]{2,}/g, '[email ocultado]')
    .replace(/\b\d{3,}\b/g, '[numero ocultado]')
    .slice(0, 240)
}
async function findCandidateRows(page: Page): Promise<Locator[]> {
  const result: Locator[] = []

  for (const frame of page.frames()) {
    const rows = frame.locator('tr, div.row')

    const rowCount = await rows.count().catch(() => 0)

    for (let rowIndex = 0; rowIndex < rowCount; rowIndex += 1) {
      const row = rows.nth(rowIndex)

      const tableCellCount = await row
        .locator(':scope > td')
        .count()
        .catch(() => 0)

      const gridCellCount = await row
        .locator(':scope > div[class*="cell"]')
        .count()
        .catch(() => 0)

      const cellCount = tableCellCount + gridCellCount

      const checkboxCount = await row
        .locator('input[type="checkbox"]')
        .count()
        .catch(() => 0)

      const actionCount = await row
        .locator(
          [
            'a',
            'button',
            'input[type="button"]',
            'input[type="submit"]',
            '[role="button"]',
            'i[title]'
          ].join(',')
        )
        .count()
        .catch(() => 0)

      const isResearchRow = cellCount >= 6 && checkboxCount > 0 && actionCount > 0

      if (isResearchRow) {
        result.push(row)
      }
    }
  }

  return result
}

async function inspectRow(row: Locator): Promise<Record<string, unknown>> {
  const tableCells = row.locator(':scope > td')

  const gridCells = row.locator(':scope > div[class*="cell"]')

  const tableCellCount = await tableCells.count()

  const cells = tableCellCount > 0 ? tableCells : gridCells

  const cellCount = await cells.count()

  const cellSummary: Array<Record<string, unknown>> = []

  for (let index = 0; index < cellCount; index += 1) {
    const cell = cells.nth(index)

    cellSummary.push({
      index,

      className: (await cell.getAttribute('class')) ?? '',

      interactiveCount: await cell
        .locator(['a', 'button', 'input', '[role="button"]', 'i[title]'].join(','))
        .count(),

      textLength: (await cell.innerText().catch(() => '')).replace(/\s+/g, ' ').trim().length
    })
  }

  const rawElements = await row
    .locator(['a', 'button', 'input', '[role="button"]', 'i', 'img'].join(','))
    .evaluateAll((elements): RawElement[] =>
      elements.map((element) => {
        const htmlElement = element as HTMLElement

        const inputElement = element as HTMLInputElement

        const anchorElement = element as HTMLAnchorElement

        return {
          tag: element.tagName.toLowerCase(),

          id: htmlElement.id ?? '',

          name: inputElement.name ?? '',

          type: inputElement.type ?? '',

          className: typeof htmlElement.className === 'string' ? htmlElement.className : '',

          title: htmlElement.getAttribute('title') ?? '',

          role: htmlElement.getAttribute('role') ?? '',

          href: anchorElement.href ?? '',

          onclick: htmlElement.getAttribute('onclick') ?? '',

          target: anchorElement.target ?? '',

          text: htmlElement.innerText ?? htmlElement.textContent ?? '',

          value: inputElement.value ?? ''
        }
      })
    )

  const controls = rawElements.map((element) => ({
    tag: element.tag,

    id: sanitizeAttribute(element.id),

    name: sanitizeAttribute(element.name),

    type: element.type,

    className: element.className,

    title: sanitizeAttribute(element.title),

    role: element.role,

    hrefPath: sanitizeUrl(element.href),

    onclick: sanitizeJavascript(element.onclick),

    target: element.target,

    text: sanitizeText(element.text),

    value:
      element.type === 'button' || element.type === 'submit'
        ? sanitizeText(element.value)
        : element.value
          ? '[ocultado]'
          : ''
  }))

  return {
    structure: tableCellCount > 0 ? 'TABLE_ROW' : 'DIV_GRID_ROW',

    tag: await row.evaluate((element) => element.tagName.toLowerCase()),

    id: (await row.getAttribute('id')) ?? '',

    className: (await row.getAttribute('class')) ?? '',

    cellCount,
    cells: cellSummary,
    controls
  }
}

async function main(): Promise<void> {
  const appData = process.env.APPDATA

  if (!appData) {
    throw new Error('Variável APPDATA não encontrada.')
  }

  const terminal = createInterface({
    input: stdin,
    output: stdout
  })

  const abortController = new AbortController()

  const handleSigint = (): void => {
    console.log('\nParada solicitada.')

    abortController.abort()
  }

  process.once('SIGINT', handleSigint)

  const context = await chromium.launchPersistentContext(
    join(appData, 'AgenteCPI', 'browser-profile'),
    {
      headless: false,
      viewport: null,
      acceptDownloads: true,
      args: ['--start-maximized']
    }
  )

  try {
    const pageManager = new PlatformPageManager()

    await pageManager.ensurePages(context)

    let cpiPage = findCpiPage(context.pages())

    if (!cpiPage) {
      throw new Error('A aba da CPI Central não foi localizada.')
    }

    if (!cpiPage.url().toLowerCase().includes('/mod_t_aguardando.aspx')) {
      await terminal.question(
        'Faça o login manual na CPI, deixe a fila aberta e pressione ENTER...'
      )

      cpiPage = findCpiPage(context.pages())
    }

    if (!cpiPage || !cpiPage.url().toLowerCase().includes('/mod_t_aguardando.aspx')) {
      throw new Error('A fila autenticada da CPI não foi confirmada.')
    }

    await cpiPage.bringToFront()

    const filterController = new CpiQueueFilterController()

    const availabilityReader = new CpiQueueAvailabilityReader()

    const monitor = new CpiQueueMonitor(filterController, availabilityReader)

    console.log('Aplicando filtro 29...')

    await filterController.applyCriminalForumFilter(cpiPage)

    console.log('Aguardando uma pesquisa disponível.')

    const monitorResult = await monitor.waitForResearch(cpiPage, {
      signal: abortController.signal,

      onState: (state, checks) => {
        const time = new Date().toLocaleTimeString('pt-BR')

        console.log(`[${time}] Verificação ${checks}: ${state}`)
      }
    })

    if (monitorResult.status === 'PAUSED') {
      console.log('Inspeção interrompida.')

      return
    }

    console.log('Pesquisa detectada. Inspecionando somente a primeira linha...')

    const candidateRows = await findCandidateRows(cpiPage)

    if (candidateRows.length === 0) {
      throw new Error('Nenhuma linha interativa de pesquisa foi encontrada.')
    }

    const report = {
      platform: 'CPI Central',
      pageUrl: sanitizeUrl(cpiPage.url()),
      candidateRowCount: candidateRows.length,
      firstRow: await inspectRow(candidateRows[0])
    }

    console.dir(report, {
      depth: null,
      colors: true
    })

    console.log('Nenhuma pesquisa foi aberta ou alterada.')

    await terminal.question('Pressione ENTER para fechar o navegador com segurança...')
  } catch (error: unknown) {
    console.error('Falha ao inspecionar a primeira pesquisa:', error)

    await terminal.question('O navegador permanecerá aberto. Pressione ENTER para encerrar...')

    process.exitCode = 1
  } finally {
    process.removeListener('SIGINT', handleSigint)

    terminal.close()
    await context.close()

    console.log('Navegador encerrado com segurança.')
  }
}

async function inspectForms(frame: Frame): Promise<Array<Record<string, unknown>>> {
  const rawForms = await frame.locator('form').evaluateAll((elements): RawForm[] =>
    elements.map((element) => {
      const form = element as HTMLFormElement

      return {
        id: form.id ?? '',
        name: form.name ?? '',
        method: form.method ?? '',
        action: form.action ?? ''
      }
    })
  )

  return rawForms.map((form) => ({
    id: form.id,
    name: form.name,
    method: form.method,
    actionPath: sanitizeUrl(form.action)
  }))
}

async function inspectControls(frame: Frame): Promise<Array<Record<string, unknown>>> {
  const rawControls = await frame
    .locator('input, select, textarea, button, a')
    .evaluateAll((elements): RawControl[] =>
      elements.map((element) => {
        const htmlElement = element as HTMLElement

        const input = element as HTMLInputElement

        const select = element as HTMLSelectElement

        let associatedLabel = ''

        if (htmlElement.id) {
          const label = document.querySelector(`label[for="${CSS.escape(htmlElement.id)}"]`)

          associatedLabel = label?.textContent ?? ''
        }

        if (!associatedLabel) {
          associatedLabel = element.closest('label')?.textContent ?? ''
        }

        const optionTexts =
          element.tagName === 'SELECT'
            ? Array.from(select.options).map((option) => option.textContent ?? '')
            : []

        const selectedOptionText =
          element.tagName === 'SELECT' ? (select.selectedOptions[0]?.textContent ?? '') : ''

        return {
          tag: element.tagName.toLowerCase(),
          id: htmlElement.id ?? '',
          name: input.name ?? '',
          type: input.type ?? '',
          className: typeof htmlElement.className === 'string' ? htmlElement.className : '',
          title: htmlElement.getAttribute('title') ?? '',
          ariaLabel: htmlElement.getAttribute('aria-label') ?? '',
          placeholder: input.placeholder ?? '',
          associatedLabel,
          disabled: input.disabled ?? false,
          readOnly: input.readOnly ?? false,
          required: input.required ?? false,
          valueLength: typeof input.value === 'string' ? input.value.length : 0,
          optionTexts,
          selectedOptionText
        }
      })
    )

  return rawControls.map((control) => ({
    tag: control.tag,
    id: sanitizeLabel(control.id),
    name: sanitizeLabel(control.name),
    type: control.type,
    className: control.className,
    title: sanitizeLabel(control.title),
    ariaLabel: sanitizeLabel(control.ariaLabel),
    placeholder: sanitizeLabel(control.placeholder),
    associatedLabel: sanitizeLabel(control.associatedLabel),
    disabled: control.disabled,
    readOnly: control.readOnly,
    required: control.required,
    hasValue: control.valueLength > 0,
    valueLength: control.valueLength,
    optionCount: control.optionTexts.length,
    optionTexts: control.optionTexts.map(sanitizeOption),
    selectedOptionText: sanitizeOption(control.selectedOptionText)
  }))
}

async function readAssociatedLabel(page: Page, control: Locator, id: string): Promise<string> {
  if (id) {
    const escapedId = id.replace(/["\\]/g, '\\$&')

    const associatedLabel = page.locator(`label[for="${escapedId}"]`).first()

    if ((await associatedLabel.count()) > 0) {
      return associatedLabel
        .innerText({
          timeout: 500
        })
        .catch(() => '')
    }
  }

  const wrappingLabel = control.locator('xpath=ancestor::label[1]').first()

  if ((await wrappingLabel.count()) > 0) {
    return wrappingLabel
      .innerText({
        timeout: 500
      })
      .catch(() => '')
  }

  return ''
}

async function inspectControl(
  page: Page,
  control: Locator,
  tag: string
): Promise<InspectedControl | null> {
  if (!(await control.isVisible())) {
    return null
  }

  const id = (await control.getAttribute('id')) ?? ''

  const form = control.locator('xpath=ancestor::form[1]')

  const formExists = (await form.count()) > 0

  const associatedLabel = await readAssociatedLabel(page, control, id)

  const text =
    tag === 'button' || tag === 'a'
      ? await control.innerText().catch(() => '')
      : ((await control.getAttribute('value')) ?? '')

  return {
    tag,
    id: sanitizeText(id),
    name: sanitizeText((await control.getAttribute('name')) ?? ''),
    type: (await control.getAttribute('type')) ?? '',
    className: (await control.getAttribute('class')) ?? '',
    placeholder: sanitizeText((await control.getAttribute('placeholder')) ?? ''),
    ariaLabel: sanitizeText((await control.getAttribute('aria-label')) ?? ''),
    title: sanitizeText((await control.getAttribute('title')) ?? ''),
    associatedLabel: sanitizeText(associatedLabel),
    formId: formExists ? sanitizeText((await form.getAttribute('id')) ?? '') : '',
    formName: formExists ? sanitizeText((await form.getAttribute('name')) ?? '') : '',
    text: tag === 'input' ? '' : sanitizeText(text)
  }
}

async function inspectPage(page: Page): Promise<Record<string, unknown>> {
  const controls: InspectedControl[] = []

  const groups = [
    {
      tag: 'input',
      selector: 'input'
    },
    {
      tag: 'select',
      selector: 'select'
    },
    {
      tag: 'textarea',
      selector: 'textarea'
    },
    {
      tag: 'button',
      selector: 'button'
    },
    {
      tag: 'a',
      selector: 'a'
    }
  ]

  for (const group of groups) {
    const locator = page.locator(group.selector)

    const count = Math.min(await locator.count(), 100)

    for (let index = 0; index < count; index += 1) {
      const inspected = await inspectControl(page, locator.nth(index), group.tag)

      if (!inspected) {
        continue
      }

      const searchable = [
        inspected.id,
        inspected.name,
        inspected.placeholder,
        inspected.ariaLabel,
        inspected.title,
        inspected.associatedLabel,
        inspected.text
      ]
        .join(' ')
        .toLowerCase()

      if (
        /cpf|documento|pessoa|consultar|consulta|buscar|pesquisar|nome/.test(searchable) ||
        inspected.tag === 'input'
      ) {
        controls.push(inspected)
      }
    }
  }

  const forms: Array<Record<string, string>> = []

  const formLocator = page.locator('form')

  const formCount = await formLocator.count()

  for (let index = 0; index < formCount; index += 1) {
    const form = formLocator.nth(index)

    forms.push({
      id: sanitizeText((await form.getAttribute('id')) ?? ''),
      name: sanitizeText((await form.getAttribute('name')) ?? ''),
      method: (await form.getAttribute('method')) ?? '',
      actionPath: sanitizeUrl((await form.getAttribute('action')) ?? '')
    })
  }

  return {
    platform: 'DeskData',
    title: await page.title(),
    url: sanitizeUrl(page.url()),
    forms,
    controls
  }
}

}

export interface PlatformPagesSnapshot {
  cpiPageUrl: string
  deskDataPageUrl: string
}

export class PlatformPageManager {
  private readonly urls: PlatformUrls

  constructor(urlInput: unknown = undefined) {
    this.urls = urlInput === undefined ? parsePlatformUrls() : parsePlatformUrls(urlInput)
  }

  async ensurePages(context: ManagedBrowserContext): Promise<PlatformPagesSnapshot> {
    const usedPages = new Set<ManagedPage>()

    const cpiPage = await this.resolvePage(context, this.urls.cpiQueueUrl, usedPages)

    usedPages.add(cpiPage)

    const deskDataPage = await this.resolvePage(context, this.urls.deskDataPeopleUrl, usedPages)

    await cpiPage.bringToFront()

    return {
      cpiPageUrl: cpiPage.url(),
      deskDataPageUrl: deskDataPage.url()
    }
  }

  private async resolvePage(
    context: ManagedBrowserContext,
    targetUrl: string,
    usedPages: ReadonlySet<ManagedPage>
  ): Promise<ManagedPage> {
    const availablePages = context
      .pages()
      .filter((page) => !page.isClosed() && !usedPages.has(page))

    const matchingPage = availablePages.find((page) => this.hasSameHostname(page.url(), targetUrl))

    const blankPage = availablePages.find((page) => page.url() === 'about:blank')

    const page = matchingPage ?? blankPage ?? (await context.newPage())

    if (page.url() !== targetUrl) {
      await page.goto(targetUrl, {
        waitUntil: 'domcontentloaded',
        timeout: 45_000
      })
    }

    return page
  }

  private hasSameHostname(currentUrl: string, targetUrl: string): boolean {
    try {
      return new URL(currentUrl).hostname === new URL(targetUrl).hostname
    } catch {
      return false
    }
  }
}

import { setTimeout as wait } from 'node:timers/promises'
import type { Page } from 'playwright'
import type { CpiQueueAvailabilityState } from './cpi-queue-availability'

export interface CpiQueueRefresher {
  refreshQueue(page: Page): Promise<void>
}

export interface CpiQueueAvailabilitySource {
  read(page: Page): Promise<CpiQueueAvailabilityState>
}

export interface CpiQueueDelay {
  wait(milliseconds: number, signal?: AbortSignal): Promise<void>
}

export interface CpiQueueMonitorOptions {
  isPauseRequested?: () => boolean
  signal?: AbortSignal
  onState?: (state: CpiQueueAvailabilityState, checks: number) => void
}

export interface CpiQueueMonitorResult {
  status: 'RESEARCH_AVAILABLE' | 'PAUSED'
  checks: number
}

class NodeQueueDelay implements CpiQueueDelay {
  async wait(milliseconds: number, signal?: AbortSignal): Promise<void> {
    await wait(milliseconds, undefined, {
      signal
    })
  }
}

export class CpiQueueMonitor {
  constructor(
    private readonly refresher: CpiQueueRefresher,
    private readonly availabilitySource: CpiQueueAvailabilitySource,
    private readonly delay: CpiQueueDelay = new NodeQueueDelay(),
    private readonly intervalMs = 60_000
  ) {
    if (intervalMs < 30_000) {
      throw new Error('Intervalo da fila deve ser de pelo menos 30000 ms')
    }
  }

  async waitForResearch(
    page: Page,
    options: CpiQueueMonitorOptions = {}
  ): Promise<CpiQueueMonitorResult> {
    let checks = 0

    while (true) {
      if (options.signal?.aborted || options.isPauseRequested?.()) {
        return {
          status: 'PAUSED',
          checks
        }
      }

      await this.refresher.refreshQueue(page)

      checks += 1

      const state = await this.availabilitySource.read(page)

      options.onState?.(state, checks)

      if (state === 'HAS_RESEARCH') {
        return {
          status: 'RESEARCH_AVAILABLE',
          checks
        }
      }

      if (state === 'INDETERMINATE') {
        throw new Error('Não foi possível determinar o estado da fila da CPI')
      }

      if (options.signal?.aborted || options.isPauseRequested?.()) {
        return {
          status: 'PAUSED',
          checks
        }
      }

      try {
        await this.delay.wait(this.intervalMs, options.signal)
      } catch (error: unknown) {
        if (options.signal?.aborted) {
          return {
            status: 'PAUSED',
            checks
          }
        }

        throw error
      }
    }
  }
}

export const CPI_RESEARCH_PDF_UPLOAD_SELECTOR = '#MainPage_FileUpload1'

type AttachmentState = 'IDLE' | 'ATTACHING' | 'ATTACHED'

interface AttachedFileState {
  count: number
  name: string
}

export interface CpiResearchPdfAttachmentControllerOptions {
  controlSearchAttempts?: number
  controlSearchIntervalMs?: number
}

export interface CpiResearchPdfAttachmentResult {
  attached: true
  filePath: string
  fileName: string
  sizeBytes: number
}

export class CpiResearchPdfAttachmentController {
  private readonly controlSearchAttempts: number
  private readonly controlSearchIntervalMs: number
  private state: AttachmentState = 'IDLE'

  constructor(options: CpiResearchPdfAttachmentControllerOptions = {}) {
    this.controlSearchAttempts = options.controlSearchAttempts ?? 40

    this.controlSearchIntervalMs = options.controlSearchIntervalMs ?? 250

    if (this.controlSearchAttempts < 1) {
      throw new Error('O número de tentativas do campo de anexo deve ser maior que zero.')
    }

    if (this.controlSearchIntervalMs < 0) {
      throw new Error('O intervalo do campo de anexo não pode ser negativo.')
    }
  }

  async attachOnce(frame: Frame, filePath: string): Promise<CpiResearchPdfAttachmentResult> {
    if (this.state !== 'IDLE') {
      throw new Error('O PDF já foi anexado ou está sendo anexado nesta execução.')
    }

    this.state = 'ATTACHING'

    try {
      const sizeBytes = await this.validatePdf(filePath)
      const fileName = basename(filePath)

      const uploadControl = await this.waitForUploadControl(frame)

      await uploadControl.setInputFiles(filePath)

      const attachedState = await uploadControl.evaluate<AttachedFileState>((element) => {
        const input = element as HTMLInputElement

        return {
          count: input.files?.length ?? 0,
          name: input.files?.item(0)?.name ?? ''
        }
      })

      if (attachedState.count !== 1) {
        throw new Error('A CPI não confirmou exatamente um arquivo no campo de anexo.')
      }

      if (attachedState.name !== fileName) {
        throw new Error('O arquivo confirmado no campo da CPI não corresponde ao PDF selecionado.')
      }

      this.state = 'ATTACHED'

      return {
        attached: true,
        filePath,
        fileName,
        sizeBytes
      }
    } catch (error: unknown) {
      this.state = 'IDLE'

      throw error
    }
  }

  private async waitForUploadControl(frame: Frame): Promise<Locator> {
    const uploadControl = frame.locator(CPI_RESEARCH_PDF_UPLOAD_SELECTOR)

    for (let attempt = 0; attempt < this.controlSearchAttempts; attempt += 1) {
      const count = await uploadControl.count()

      if (count === 1) {
        return uploadControl
      }

      if (count > 1) {
        throw new Error(
          'O campo de anexo deveria possuir exatamente um controle. Encontrados: ' + count
        )
      }

      if (attempt < this.controlSearchAttempts - 1) {
        await wait(this.controlSearchIntervalMs)
      }
    }

    throw new Error('O campo de anexo deveria possuir exatamente um controle. Encontrados: 0')
  }

  private async validatePdf(filePath: string): Promise<number> {
    if (extname(filePath).toLowerCase() !== '.pdf') {
      throw new Error('O arquivo selecionado não possui extensão PDF.')
    }

    const fileStat = await stat(filePath)

    if (!fileStat.isFile() || fileStat.size < 5) {
      throw new Error('O arquivo PDF selecionado está vazio ou inválido.')
    }

    const signature = Buffer.alloc(5)
    const fileHandle = await open(filePath, 'r')

    try {
      const readResult = await fileHandle.read(signature, 0, signature.length, 0)

      if (readResult.bytesRead !== signature.length || signature.toString('ascii') !== '%PDF-') {
        throw new Error('O arquivo selecionado não possui uma assinatura PDF válida.')
      }
    } finally {
      await fileHandle.close()
    }

    return fileStat.size
  }
}

      throw new Error('Número de tentativas deve ser maior que zero.')
    }

    if (this.resultSearchIntervalMs < 0) {
      throw new Error('Intervalo de busca não pode ser negativo.')
    }
  }

  async prepare(
    frame: CpiResearchResultFrame,
    classification: ResearchClassification
  ): Promise<PreparedCpiResearchResult> {
    const resultOption = mapResearchClassificationToCpiResult(classification)

    await this.selectByLabel(frame, RESEARCHER_SELECTOR, RESEARCHER_LABEL, 'Pesquisador/Fonte')

    await this.selectByLabel(
      frame,
      PRIMARY_RESEARCHER_SELECTOR,
      RESEARCHER_LABEL,
      'Pesquisador/Fonte 1º'
    )

    await this.selectByLabel(frame, STATUS_SELECTOR, STATUS_LABEL, 'Status')

    const resultRadio = await this.waitForSingleControl(
      frame,
      resultOption.selector,
      'Opção de resultado da CPI'
    )

    const resultId = resultOption.selector.replace(/^#/, '')

    const resultLabel = await this.waitForSingleControl(
      frame,
      'label[for="' + resultId + '"]',
      'Rótulo visível da opção de resultado da CPI'
    )

    if (!resultLabel.click) {
      throw new Error('O rótulo da opção de resultado não permite clique.')
    }

    await resultLabel.click()

    if (!resultRadio.isChecked) {
      throw new Error('Não foi possível confirmar a opção selecionada.')
    }

    if (!(await resultRadio.isChecked())) {
      throw new Error('A opção de resultado da CPI não ficou selecionada.')
    }

    return {
      classification,
      selectedResultLabel: resultOption.label,
      researcherLabel: RESEARCHER_LABEL,
      primaryResearcherLabel: RESEARCHER_LABEL,
      statusLabel: STATUS_LABEL,
      responseChanged: false,
      saved: false
    }
  }

  private async selectByLabel(
    frame: CpiResearchResultFrame,
    selector: string,
    label: string,
    fieldName: string
  ): Promise<void> {
    const control = await this.waitForSingleControl(frame, selector, fieldName)

    if (!control.selectOption) {
      throw new Error(fieldName + ' não permite seleção.')
    }

    await control.selectOption({ label })
  }

  private async waitForSingleControl(
    frame: CpiResearchResultFrame,
    selector: string,
    fieldName: string
  ): Promise<CpiResearchResultLocator> {
    const control = frame.locator(selector)

    for (let attempt = 0; attempt < this.resultSearchAttempts; attempt += 1) {
      const count = await control.count()

      if (count === 1) {
        return control
      }

      if (count > 1) {
        throw new Error(
          fieldName + ' deveria possuir exatamente um controle. Encontrados: ' + count
        )
      }

      if (attempt < this.resultSearchAttempts - 1) {
        await wait(this.resultSearchIntervalMs)
      }
    }

    throw new Error(fieldName + ' não apareceu após aguardar a renderização.')
  }
}

} as const

export interface DeskDataDatasetLocator {
  count(): Promise<number>

  nth(index: number): DeskDataDatasetLocator

  getAttribute(name: string): Promise<string | null>

  isChecked(): Promise<boolean>

  evaluate(
    callback: (element: Element, shouldBeChecked: boolean) => void,
    shouldBeChecked: boolean
  ): Promise<void>
}

export interface DeskDataDatasetPage {
  locator(selector: string): DeskDataDatasetLocator
}

export interface DeskDataDatasetSnapshot {
  selectedIds: string[]
  selectedCount: number
}

export class DeskDataDatasetController {
  applyApprovedDatasets(page: DeskDataDatasetPage): Promise<DeskDataDatasetSnapshot> {
    return this.applyDatasetIds(page, APPROVED_DESKDATA_DATASET_IDS)
  }

  async applyDatasetIds(
    page: DeskDataDatasetPage,
    requestedIds: readonly string[]
  ): Promise<DeskDataDatasetSnapshot> {
    const selectedIds = [...new Set(requestedIds.map((id) => id.trim()).filter(Boolean))]

    if (selectedIds.length === 0) {
      throw new Error('Informe pelo menos um conjunto de dados da DeskData')
    }

    const checkboxes = page.locator(DESKDATA_DATASET_SELECTORS.allCheckboxes)

    const checkboxCount = await checkboxes.count()

    if (checkboxCount === 0) {
      throw new Error('Nenhum conjunto de dados foi encontrado na DeskData')
    }

    const selectedIdSet = new Set(selectedIds)
    const foundIds = new Set<string>()

    for (let index = 0; index < checkboxCount; index += 1) {
      const checkbox = checkboxes.nth(index)
      const checkboxId = await checkbox.getAttribute('id')

      if (!checkboxId) {
        continue
      }

      const shouldBeChecked = selectedIdSet.has(checkboxId)

      if (shouldBeChecked) {
        foundIds.add(checkboxId)
      }

      await checkbox.evaluate((element, targetState) => {
        const input = element as HTMLInputElement

        if (input.checked === targetState) {
          return
        }

        input.checked = targetState

        input.dispatchEvent(
          new Event('input', {
            bubbles: true
          })
        )

        input.dispatchEvent(
          new Event('change', {
            bubbles: true
          })
        )
      }, shouldBeChecked)

      if ((await checkbox.isChecked()) !== shouldBeChecked) {
        throw new Error(`A DeskData não manteve o estado do conjunto ${checkboxId}`)
      }
    }

    const missingIds = selectedIds.filter((id) => !foundIds.has(id))

    if (missingIds.length > 0) {
      throw new Error(
        `Conjuntos obrigatórios não encontrados na DeskData: ${missingIds.join(', ')}`
      )
    }

    return {
      selectedIds,
      selectedCount: selectedIds.length
    }
  }
}

import { mkdir, open, readdir, stat } from 'node:fs/promises'
import { extname, join } from 'node:path'

const DEFAULT_PDF_DIRECTORY = 'E:\\PDF-AGENTE-CPI'

export interface DeskDataPdfFile {
  filePath: string
  fileName: string
  sizeBytes: number
  modifiedAtMs: number
}

export class DeskDataLatestPdfLocator {
  constructor(private readonly directory = DEFAULT_PDF_DIRECTORY) {}

  async findLatestValid(): Promise<DeskDataPdfFile> {
    await mkdir(this.directory, {
      recursive: true
    })

    const entries = await readdir(this.directory, {
      withFileTypes: true
    })

    const candidates = await Promise.all(
      entries
        .filter((entry) => entry.isFile() && extname(entry.name).toLowerCase() === '.pdf')
        .map(async (entry) => {
          return this.inspectFile(join(this.directory, entry.name), entry.name)
        })
    )

    const validCandidates = candidates
      .filter((candidate): candidate is DeskDataPdfFile => candidate !== undefined)
      .sort((left, right) => right.modifiedAtMs - left.modifiedAtMs)

    const latest = validCandidates.at(0)

    if (!latest) {
      throw new Error('Nenhum PDF válido da DeskData foi encontrado em ' + this.directory + '.')
    }

    return latest
  }

  private async inspectFile(
    filePath: string,
    fileName: string
  ): Promise<DeskDataPdfFile | undefined> {
    try {
      const fileStat = await stat(filePath)

      if (!fileStat.isFile() || fileStat.size < 5) {
        return undefined
      }

      const signature = Buffer.alloc(5)
      const fileHandle = await open(filePath, 'r')

      try {
        const readResult = await fileHandle.read(signature, 0, signature.length, 0)

        if (readResult.bytesRead !== signature.length || signature.toString('ascii') !== '%PDF-') {
          return undefined
        }
      } finally {
        await fileHandle.close()
      }

      return {
        filePath,
        fileName,
        sizeBytes: fileStat.size,
        modifiedAtMs: fileStat.mtimeMs
      }
    } catch {
      return undefined
    }
  }
}

import { randomBytes } from 'node:crypto'
import { mkdir, open, rm, stat } from 'node:fs/promises'
import { extname, join } from 'node:path'
import type { BrowserContext, Download, Page } from 'playwright'

export const DESKDATA_PDF_EXPORT_SELECTOR =
  'form[method="POST"] input[type="submit"][name="export"][value="Exportar"][formtarget="_blank"]'

const DEFAULT_OUTPUT_DIRECTORY = 'E:\\PDF-AGENTE-CPI'

type DownloadState = 'IDLE' | 'DOWNLOADING' | 'DOWNLOADED'

export interface DeskDataPdfDownloadControllerOptions {
  outputDirectory?: string
  downloadTimeoutMs?: number
}

export interface DeskDataPdfDownloadResult {
  filePath: string
  sizeBytes: number
}

export class DeskDataPdfDownloadController {
  private readonly outputDirectory: string
  private readonly downloadTimeoutMs: number
  private state: DownloadState = 'IDLE'

  constructor(options: DeskDataPdfDownloadControllerOptions = {}) {
    this.outputDirectory = options.outputDirectory ?? DEFAULT_OUTPUT_DIRECTORY
    this.downloadTimeoutMs = options.downloadTimeoutMs ?? 30_000

    if (this.downloadTimeoutMs < 1) {
      throw new Error('O tempo limite do download deve ser maior que zero.')
    }
  }

  async downloadOnce(
    context: BrowserContext,
    sourcePage: Page
  ): Promise<DeskDataPdfDownloadResult> {
    if (this.state !== 'IDLE') {
      throw new Error('O PDF da DeskData já foi exportado ou está sendo exportado nesta execução.')
    }

    const exportControl = sourcePage.locator(DESKDATA_PDF_EXPORT_SELECTOR)

    const exportControlCount = await exportControl.count()

    if (exportControlCount !== 1) {
      throw new Error(
        'O botão Exportar deveria possuir exatamente um controle. Encontrados: ' +
          exportControlCount
      )
    }

    this.state = 'DOWNLOADING'

    let savedFilePath: string | undefined
    let timeoutHandle: ReturnType<typeof setTimeout> | undefined

    let resolveDownload: (download: Download) => void

    const downloadPromise = new Promise<Download>((resolve) => {
      resolveDownload = resolve
    })

    let downloadCaptured = false

    const onDownload = (download: Download): void => {
      if (downloadCaptured) {
        return
      }

      downloadCaptured = true
      resolveDownload(download)
    }

    const registeredPages = new Set<Page>()

    const registerPage = (page: Page): void => {
      if (registeredPages.has(page)) {
        return
      }

      registeredPages.add(page)
      page.on('download', onDownload)
    }

    const onPage = (page: Page): void => {
      registerPage(page)
    }

    context.pages().forEach(registerPage)
    registerPage(sourcePage)
    context.on('page', onPage)

    try {
      const timeoutPromise = new Promise<never>((_resolve, reject) => {
        timeoutHandle = setTimeout(() => {
          reject(new Error('O download do PDF da DeskData não iniciou dentro do tempo limite.'))
        }, this.downloadTimeoutMs)
      })

      await exportControl.click({
        timeout: 10_000
      })

      const download = await Promise.race([downloadPromise, timeoutPromise])

      const downloadFailure = await download.failure()

      if (downloadFailure) {
        throw new Error('A DeskData informou falha no download do PDF: ' + downloadFailure)
      }

      await mkdir(this.outputDirectory, {
        recursive: true
      })

      savedFilePath = join(this.outputDirectory, this.createTechnicalFileName())

      await download.saveAs(savedFilePath)

      const sizeBytes = await this.validatePdf(savedFilePath)

      this.state = 'DOWNLOADED'

      return {
        filePath: savedFilePath,
        sizeBytes
      }
    } catch (error: unknown) {
      this.state = 'IDLE'

      if (savedFilePath) {
        await rm(savedFilePath, {
          force: true
        })
      }

      throw error
    } finally {
      if (timeoutHandle) {
        clearTimeout(timeoutHandle)
      }

      context.off('page', onPage)

      registeredPages.forEach((page) => {
        page.off('download', onDownload)
      })
    }
  }

  private createTechnicalFileName(): string {
    const timestamp = new Date()
      .toISOString()
      .replace(/\.\d{3}Z$/, '')
      .replace(/[-:]/g, '')
      .replace('T', '-')

    const token = randomBytes(4).toString('hex')

    return 'deskdata-' + timestamp + '-' + token + '.pdf'
  }

  private async validatePdf(filePath: string): Promise<number> {
    if (extname(filePath).toLowerCase() !== '.pdf') {
      throw new Error('O arquivo baixado não possui extensão PDF.')
    }

    const fileStat = await stat(filePath)

    if (!fileStat.isFile() || fileStat.size < 5) {
      throw new Error('O arquivo PDF baixado está vazio ou inválido.')
    }

    const signature = Buffer.alloc(5)
    const fileHandle = await open(filePath, 'r')

    try {
      const readResult = await fileHandle.read(signature, 0, signature.length, 0)

      if (readResult.bytesRead !== signature.length || signature.toString('ascii') !== '%PDF-') {
        throw new Error('O arquivo baixado não possui uma assinatura PDF válida.')
      }
    } finally {
      await fileHandle.close()
    }

    return fileStat.size
  }
}

const PROCESS_EVIDENCE_PATTERN =
  /\b(process|processo|processos|criminal|penal|cível|civel|civil|trabalhista|inquérito|inquerito|ação|acao|execução|execucao|tribunal|vara|comarca)\b/i

const PROCESS_ABSENCE_PATTERN =
  /(nenhum|nenhuma|não|nao|sem).{0,50}(process|registro|resultado|informação|informacao)/i

const PROCESS_NUMBER_PATTERN = /\b\d{4,7}-\d{2}\.\d{4}\.\d\.\d{2}\.\d{4}\b/

export class DeskDataProcessDomCapture {
  async captureTables(page: DeskDataProcessCapturePage): Promise<RawDeskDataInspectionTable[]> {
    const locator = page.locator(DESKDATA_PROCESS_CAPTURE_SELECTORS.tables)

    const count = Math.min(await locator.count(), 30)

    const tables: RawDeskDataInspectionTable[] = []

    for (let index = 0; index < count; index += 1) {
      const table = locator.nth(index)

      const text = await table
        .innerText({
          timeout: 2_000
        })
        .catch(() => '')

      if (!text.trim()) {
        continue
      }

      tables.push({
        index,
        text,
        html: ''
      })
    }

    return tables
  }

  async captureCandidates(
    page: DeskDataProcessCapturePage
  ): Promise<RawDeskDataInspectionCandidate[]> {
    const locator = page.locator(DESKDATA_PROCESS_CAPTURE_SELECTORS.candidates)

    const count = Math.min(await locator.count(), 150)

    const candidates: RawDeskDataInspectionCandidate[] = []

    for (let index = 0; index < count; index += 1) {
      const candidate = locator.nth(index)

      const text = await candidate
        .innerText({
          timeout: 2_000
        })
        .catch(() => '')

      if (!text.trim() || !this.containsProcessEvidence(text)) {
        continue
      }

      const [id, className] = await Promise.all([
        candidate.getAttribute('id').catch(() => null),

        candidate.getAttribute('class').catch(() => null)
      ])

      candidates.push({
        index,
        tagName: '',
        id: id ?? '',
        className: className ?? '',
        text,
        html: ''
      })
    }

    return candidates
  }

  private containsProcessEvidence(text: string): boolean {
    return (
      PROCESS_EVIDENCE_PATTERN.test(text) ||
      PROCESS_ABSENCE_PATTERN.test(text) ||
      PROCESS_NUMBER_PATTERN.test(text)
    )
  }

import type { DeskDataResultSummary } from '../../domain/research/classify-deskdata-result'
import type { DeskDataProcessInspection } from './deskdata-process-result-inspector'

const PROCESS_NUMBER_PATTERN = /\b\d{4,7}-\d{2}\.\d{4}\.\d\.\d{2}\.\d{4}\b/i

const CRIMINAL_NATURE_PATTERN =
  /\b(criminal|penal|ação penal|acao penal|execução penal|execucao penal|inquérito|inquerito|juizado especial criminal)\b/i

const NON_CRIMINAL_NATURE_PATTERN =
  /\b(cível|civel|civil|trabalhista|família|familia|consumidor|fazenda pública|fazenda publica|execução fiscal|execucao fiscal)\b/i

const NO_PROCESSES_PATTERN = /\b(nenhum|nenhuma|não|nao|sem)\b.{0,80}\b(processo|processos)\b/i

export class DeskDataProcessResultAnalyzer {
  analyze(inspection: DeskDataProcessInspection): DeskDataResultSummary {
    if (!inspection.sourcePath.startsWith('/pessoas/resultado/')) {
      return {
        hasProcesses: false,
        hasCriminalProcesses: false,
        isProcessNatureIdentifiable: false,
        isResultValid: false
      }
    }

    const capturedText = this.collectCapturedText(inspection)

    const hasProcessNumbers = PROCESS_NUMBER_PATTERN.test(capturedText)
    const explicitlyHasNoProcesses = NO_PROCESSES_PATTERN.test(capturedText)

    const hasProcesses = hasProcessNumbers && !explicitlyHasNoProcesses

    if (!hasProcesses) {
      return {
        hasProcesses: false,
        hasCriminalProcesses: false,
        isProcessNatureIdentifiable: true,
        isResultValid: true
      }
    }

    const hasCriminalProcesses = CRIMINAL_NATURE_PATTERN.test(capturedText)
    const hasNonCriminalProcesses = NON_CRIMINAL_NATURE_PATTERN.test(capturedText)

    return {
      hasProcesses: true,
      hasCriminalProcesses,
      isProcessNatureIdentifiable: hasCriminalProcesses || hasNonCriminalProcesses,
      isResultValid: true
    }
  }

  private collectCapturedText(inspection: DeskDataProcessInspection): string {
    return [
      inspection.pageTitle,
      ...inspection.relevantBodyLines,
      ...inspection.headings,
      ...inspection.tables.flatMap((table) => [table.text, table.html]),
      ...inspection.candidates.flatMap((candidate) => [candidate.text, candidate.html])
    ].join('\n')
  }
}

export type DeskDataInspectionScenario =
  'caso-a-criminal' | 'caso-b-sem-processos' | 'caso-c-nao-criminal'

export interface RawDeskDataInspectionTable {
  index: number
  text: string
  html: string
}

export interface RawDeskDataInspectionCandidate {
  index: number
  tagName: string
  id: string
  className: string
  text: string
  html: string
}

export interface RawDeskDataProcessInspection {
  scenario: DeskDataInspectionScenario
  capturedAt: string
  sourceUrl: string
  pageTitle: string
  bodyText: string
  headings: string[]
  tables: RawDeskDataInspectionTable[]
  candidates: RawDeskDataInspectionCandidate[]
}

export interface DeskDataInspectionTable {
  index: number
  text: string
  html: string
}

export interface DeskDataInspectionCandidate {
  index: number
  tagName: string
  id: string
  className: string
  text: string
  html: string
}

export interface DeskDataProcessInspection {
  schemaVersion: 1
  scenario: DeskDataInspectionScenario
  capturedAt: string
  sourcePath: string
  pageTitle: string
  relevantBodyLines: string[]
  headings: string[]
  tables: DeskDataInspectionTable[]
  candidates: DeskDataInspectionCandidate[]
}

const PROCESS_NUMBER_PATTERN = /\b\d{4,7}-\d{2}\.\d{4}\.\d\.\d{2}\.\d{4}\b/

const RELEVANT_TEXT_PATTERN =
  /\b(process|processo|processos|criminal|penal|cível|civel|civil|trabalhista|inquérito|inquerito|ação|acao|execução|execucao|tribunal|vara|comarca)\b/i

const ABSENCE_PATTERN =
  /(nenhum|nenhuma|não|nao|sem).{0,50}(process|registro|resultado|informação|informacao)/i

export class DeskDataProcessResultInspector {
  build(raw: RawDeskDataProcessInspection): DeskDataProcessInspection {
    return {
      schemaVersion: 1,
      scenario: raw.scenario,
      capturedAt: raw.capturedAt,
      sourcePath: this.extractPath(raw.sourceUrl),
      pageTitle: this.sanitizeText(raw.pageTitle),
      relevantBodyLines: this.extractRelevantLines(raw.bodyText),
      headings: this.uniqueTexts(raw.headings),
      tables: raw.tables.slice(0, 30).map((table) => ({
        index: table.index,
        text: this.truncate(this.sanitizeText(table.text), 30_000),
        html: this.truncate(this.sanitizeHtml(table.html), 80_000)
      })),
      candidates: this.sanitizeCandidates(raw.candidates)
    }
  }

  toText(snapshot: DeskDataProcessInspection): string {
    const lines: string[] = [
      'INSPEÇÃO DE RESULTADO — DESKDATA',
      '',
      `Cenário: ${snapshot.scenario}`,
      `Capturado em: ${snapshot.capturedAt}`,
      `Caminho: ${snapshot.sourcePath}`,
      '',
      'TÍTULOS',
      ...snapshot.headings.map((heading) => `- ${heading}`),
      '',
      'LINHAS RELEVANTES',
      ...snapshot.relevantBodyLines.map((line) => `- ${line}`)
    ]

    snapshot.tables.forEach((table) => {
      lines.push('', `TABELA ${table.index}`, table.text)
    })

    snapshot.candidates.forEach((candidate) => {
      lines.push(
        '',
        `ELEMENTO ${candidate.index}`,
        `Tag: ${candidate.tagName}`,
        `ID: ${candidate.id || '[sem id]'}`,
        `Classe: ${candidate.className || '[sem classe]'}`,
        candidate.text
      )
    })

    return lines.join('\n')
  }

  private sanitizeCandidates(
    candidates: RawDeskDataInspectionCandidate[]
  ): DeskDataInspectionCandidate[] {
    const seenTexts = new Set<string>()
    const result: DeskDataInspectionCandidate[] = []

    for (const candidate of candidates) {
      const text = this.truncate(this.sanitizeText(candidate.text), 30_000)

      const signature = text.toLocaleLowerCase('pt-BR').replace(/\s+/g, ' ').trim()

      if (!signature || seenTexts.has(signature)) {
        continue
      }

      seenTexts.add(signature)

      result.push({
        index: candidate.index,
        tagName: this.sanitizeText(candidate.tagName),
        id: this.sanitizeText(candidate.id),
        className: this.sanitizeText(candidate.className),
        text,
        html: this.truncate(this.sanitizeHtml(candidate.html), 80_000)
      })

      if (result.length >= 30) {
        break
      }
    }

    return result
  }

  private extractRelevantLines(bodyText: string): string[] {
    const lines = bodyText
      .split(/\r?\n/)
      .map((line) => this.sanitizeText(line))
      .map((line) => line.replace(/\s+/g, ' ').trim())
      .filter(Boolean)
      .filter(
        (line) =>
          RELEVANT_TEXT_PATTERN.test(line) ||
          ABSENCE_PATTERN.test(line) ||
          PROCESS_NUMBER_PATTERN.test(line)
      )

    return [...new Set(lines)].slice(0, 500)
  }

  private uniqueTexts(values: string[]): string[] {
    return [
      ...new Set(
        values
          .map((value) => this.sanitizeText(value))
          .map((value) => value.replace(/\s+/g, ' ').trim())
          .filter(Boolean)
      )
    ]
  }

  private sanitizeHtml(value: string): string {
    return this.sanitizeText(value).replace(/\bvalue=(["'])[^"']*\1/gi, 'value="[REMOVIDO]"')
  }

  private sanitizeText(value: string): string {
    return value
      .replace(/\b\d{3}\.?\d{3}\.?\d{3}-?\d{2}\b/g, '[CPF_REMOVIDO]')
      .replace(/\b\d{11}\b/g, '[CPF_REMOVIDO]')
  }

  private extractPath(sourceUrl: string): string {
    try {
      return new URL(sourceUrl).pathname
    } catch {
      return ''
    }
  }

  private truncate(value: string, maximumLength: number): string {
    if (value.length <= maximumLength) {
      return value
    }

    return `${value.slice(0, maximumLength)}\n[CONTEÚDO_TRUNCADO]`
  }
}

export const DESKDATA_SUBMIT_SELECTORS = {
  searchButton: 'form#form input#search[name="submit"][type="submit"]'
} as const

export interface DeskDataSubmitLocator {
  count(): Promise<number>
  isEnabled(): Promise<boolean>

  click(options?: { timeout?: number }): Promise<void>
}

export interface DeskDataSubmitPage {
  locator(selector: string): DeskDataSubmitLocator
  url(): string

  waitForLoadState(
    state: 'domcontentloaded',
    options?: {
      timeout?: number
    }
  ): Promise<void>

  waitForTimeout(milliseconds: number): Promise<void>
}

export interface DeskDataSubmitSnapshot {
  status: 'SUBMITTED'
  resultPath: string
}

type SubmissionState = 'IDLE' | 'SUBMITTING' | 'SUBMITTED'

export class DeskDataSearchSubmitController {
  private state: SubmissionState = 'IDLE'

  async submitOnce(page: DeskDataSubmitPage): Promise<DeskDataSubmitSnapshot> {
    if (this.state !== 'IDLE') {
      throw new Error('A consulta da DeskData já foi enviada')
    }

    const searchButton = page.locator(DESKDATA_SUBMIT_SELECTORS.searchButton)

    if ((await searchButton.count()) !== 1) {
      throw new Error('Botão Consultar da DeskData não encontrado')
    }

    if (!(await searchButton.isEnabled())) {
      throw new Error('Botão Consultar da DeskData está desabilitado')
    }

    this.state = 'SUBMITTING'

    try {
      await searchButton.click({
        timeout: 10_000
      })

      await page
        .waitForLoadState('domcontentloaded', {
          timeout: 30_000
        })
        .catch(() => undefined)

      await page.waitForTimeout(1_000)

      this.state = 'SUBMITTED'

      return {
        status: 'SUBMITTED',
        resultPath: this.extractPath(page.url())
      }
    } catch (error) {
      this.state = 'IDLE'
      throw error
    }
  }

  private extractPath(pageUrl: string): string {
    try {
      return new URL(pageUrl).pathname
    } catch {
      return ''
    }
  }
}

import type { ResearchClassification } from './research-classification'

export interface DeskDataResultSummary {
  hasProcesses: boolean
  hasCriminalProcesses: boolean
  isProcessNatureIdentifiable: boolean
  isResultValid: boolean
}

export function classifyDeskDataResult(result: DeskDataResultSummary): ResearchClassification {
  if (!result.isResultValid) {
    return 'INDEFINIDO'
  }

  if (!result.hasProcesses) {
    return 'HA_INFORMACOES'
  }

  if (!result.isProcessNatureIdentifiable) {
    return 'INDEFINIDO'
  }

  if (result.hasCriminalProcesses) {
    return 'CONSTA_RESTRICAO'
  }

  return 'HA_INFORMACOES'
}
