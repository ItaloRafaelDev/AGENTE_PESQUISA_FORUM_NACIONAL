import type { BrowserContext, Page } from 'playwright'
import type { CpiResearchIdentity } from '../cpi/cpi-research-identity-reader'
import {
  CpiResearchPdfPreparationWorkflow,
  type CpiResearchPdfPreparationFrame,
  type CpiResearchPdfPreparationResult
} from '../cpi/cpi-research-pdf-preparation-workflow'
import { DeskDataCpfFormController } from '../deskdata/deskdata-cpf-form-controller'
import { DeskDataDatasetController } from '../deskdata/deskdata-dataset-controller'
import {
  DeskDataInspectionClassifier,
  type DeskDataInspectionClassification
} from '../deskdata/deskdata-inspection-classifier'
import { DeskDataLiveResultCapture } from '../deskdata/deskdata-live-result-capture'
import {
  DeskDataPdfDownloadController,
  type DeskDataPdfDownloadResult
} from '../deskdata/deskdata-pdf-download-controller'
import { DeskDataResultReadyWaiter } from '../deskdata/deskdata-result-ready-waiter'
import { DeskDataSearchSubmitController } from '../deskdata/deskdata-search-submit-controller'

export interface FullResearchPreparationDependencies {
  cpfForm: Pick<DeskDataCpfFormController, 'fillCpf'>
  dataset: Pick<DeskDataDatasetController, 'applyApprovedDatasets'>
  submit: Pick<DeskDataSearchSubmitController, 'submitOnce'>
  resultReady: Pick<DeskDataResultReadyWaiter, 'waitUntilReady'>
  resultCapture: Pick<DeskDataLiveResultCapture, 'capture'>
  classifier: Pick<DeskDataInspectionClassifier, 'classify'>
  pdfDownloader: Pick<DeskDataPdfDownloadController, 'downloadOnce'>
  cpiPreparation: Pick<CpiResearchPdfPreparationWorkflow, 'prepareWithAttachment'>
}

export interface FullResearchPreparationResult {
  maskedCpf: string
  selectedDatasetCount: number
  classification: DeskDataInspectionClassification
  pdf: DeskDataPdfDownloadResult
  cpi: CpiResearchPdfPreparationResult
  detailFrame: CpiResearchPdfPreparationFrame
}

function createDefaultDependencies(): FullResearchPreparationDependencies {
  return {
    cpfForm: new DeskDataCpfFormController(),
    dataset: new DeskDataDatasetController(),
    submit: new DeskDataSearchSubmitController(),
    resultReady: new DeskDataResultReadyWaiter(),
    resultCapture: new DeskDataLiveResultCapture(),
    classifier: new DeskDataInspectionClassifier(),
    pdfDownloader: new DeskDataPdfDownloadController(),
    cpiPreparation: new CpiResearchPdfPreparationWorkflow()
  }
}

export class FullResearchPreparationWorkflow {
  constructor(
    private readonly dependencies: FullResearchPreparationDependencies = createDefaultDependencies()
  ) {}

  async prepare(
    context: BrowserContext,
    deskDataPage: Page,
    detailFrame: CpiResearchPdfPreparationFrame,
    identity: CpiResearchIdentity
  ): Promise<FullResearchPreparationResult> {
    await this.dependencies.cpfForm.fillCpf(deskDataPage, identity.cpf)

    const datasets = await this.dependencies.dataset.applyApprovedDatasets(deskDataPage)

    await this.dependencies.submit.submitOnce(deskDataPage)

    await this.dependencies.resultReady.waitUntilReady(deskDataPage)

    const inspection = await this.dependencies.resultCapture.capture(deskDataPage)

    const classification = this.dependencies.classifier.classify(inspection)

    const pdf = await this.dependencies.pdfDownloader.downloadOnce(context, deskDataPage)

    const cpi = await this.dependencies.cpiPreparation.prepareWithAttachment(
      detailFrame,
      classification.classification,
      pdf.filePath
    )

    return {
      maskedCpf: identity.maskedCpf,
      selectedDatasetCount: datasets.selectedCount,
      classification,
      pdf,
      cpi,
      detailFrame
    }
  }
}

export interface CpiFirstResearchOpenerOptions {
  frameSearchAttempts?: number
  frameSearchIntervalMs?: number
}

export interface OpenedResearchSnapshot {
  availableRows: number
  detailFramePath: string
}

export class CpiFirstResearchOpener {
  private readonly frameSearchAttempts: number
  private readonly frameSearchIntervalMs: number

  constructor(options: CpiFirstResearchOpenerOptions = {}) {
    this.frameSearchAttempts = options.frameSearchAttempts ?? 40

    this.frameSearchIntervalMs = options.frameSearchIntervalMs ?? 250
  }

  async openFirst(page: ResearchPage): Promise<OpenedResearchSnapshot> {
    const rows = page.locator(CPI_RESEARCH_SELECTORS.researchRows)

    const availableRows = await rows.count()

    if (availableRows === 0) {
      throw new Error('Nenhuma pesquisa disponível foi encontrada na fila da CPI')
    }

    const firstRow = rows.first()

    const responderAction = firstRow.locator(CPI_RESEARCH_SELECTORS.responderAction)

    if ((await responderAction.count()) !== 1) {
      throw new Error('Ação Responder / Atualizar não encontrada na primeira pesquisa')
    }

    await responderAction.click({
      timeout: 10_000
    })

    const detailFramePath = await this.waitForDetailFrame(page)

    return {
      availableRows,
      detailFramePath
    }
  }

  private async waitForDetailFrame(page: ResearchPage): Promise<string> {
    for (let attempt = 0; attempt < this.frameSearchAttempts; attempt += 1) {
      const detailFrame = page.frames().find((frame) => {
        try {
          return new URL(frame.url()).pathname
            .toLowerCase()
            .endsWith('/zpersespecial/pesquisas/upd_cad/mod_tt_pesq_upd.aspx')
        } catch {
          return false
        }
      })

      if (detailFrame) {
        return new URL(detailFrame.url()).pathname
      }

      await page.waitForTimeout(this.frameSearchIntervalMs)
    }

    throw new Error('Formulário da pesquisa não foi aberto pela CPI')
  }
}

import { setTimeout as wait } from 'node:timers/promises'
import { isValidCpf, maskCpfForLog } from '../../shared/utils/cpf'

const DETAIL_FRAME_PATH = '/zpersespecial/pesquisas/upd_cad/mod_tt_pesq_upd.aspx'

export const CPI_RESEARCH_DATA_SELECTORS = {
  cpfValue:
    'xpath=//div[contains(concat(" ", normalize-space(@class), " "), " cell ") and contains(concat(" ", normalize-space(@class), " "), " txt ") and normalize-space(.)="CPF:"]/following-sibling::div[contains(concat(" ", normalize-space(@class), " "), " cell ") and contains(concat(" ", normalize-space(@class), " "), " field ")][1]'
} as const

export interface ResearchDataLocator {
  count(): Promise<number>
  first(): ResearchDataLocator

  innerText(options?: { timeout?: number }): Promise<string>
}

export interface ResearchDataFrame {
  url(): string

  locator(selector: string): ResearchDataLocator
}

export interface ResearchDataPage {
  frames(): ResearchDataFrame[]
}

export interface CpiResearchIdentity {
  cpf: string
  maskedCpf: string
  detailFramePath: string
}

export interface CpiResearchIdentityReaderOptions {
  searchAttempts?: number
  searchIntervalMs?: number
}

export class CpiResearchIdentityReader {
  private readonly searchAttempts: number
  private readonly searchIntervalMs: number

  constructor(options: CpiResearchIdentityReaderOptions = {}) {
    this.searchAttempts = options.searchAttempts ?? 40

    this.searchIntervalMs = options.searchIntervalMs ?? 250

    if (this.searchAttempts < 1) {
      throw new Error('Número de tentativas deve ser maior que zero')
    }

    if (this.searchIntervalMs < 0) {
      throw new Error('Intervalo de busca não pode ser negativo')
    }
  }

  async read(page: ResearchDataPage): Promise<CpiResearchIdentity> {
    let detailFrameFound = false

    for (let attempt = 0; attempt < this.searchAttempts; attempt += 1) {
      const detailFrames = page.frames().filter((frame) => this.isDetailFrame(frame))

      if (detailFrames.length > 0) {
        detailFrameFound = true
      }

      for (const detailFrame of detailFrames) {
        const cpfField = detailFrame.locator(CPI_RESEARCH_DATA_SELECTORS.cpfValue)

        const cpfFieldCount = await cpfField.count().catch(() => 0)

        if (cpfFieldCount === 0) {
          continue
        }

        const rawCpf = await cpfField.first().innerText({
          timeout: 5_000
        })

        const cpf = rawCpf.replace(/\D/g, '')

        if (!isValidCpf(cpf)) {
          throw new Error('CPF capturado na CPI é inválido')
        }

        return {
          cpf,
          maskedCpf: maskCpfForLog(cpf),
          detailFramePath: new URL(detailFrame.url()).pathname
        }
      }

      if (attempt < this.searchAttempts - 1) {
        await wait(this.searchIntervalMs)
      }
    }

    if (!detailFrameFound) {
      throw new Error('Formulário interno da pesquisa da CPI não encontrado')
    }

    throw new Error('Campo CPF da pesquisa da CPI não encontrado')
  }

  private isDetailFrame(frame: ResearchDataFrame): boolean {
    try {
      return new URL(frame.url()).pathname.toLowerCase() === DETAIL_FRAME_PATH
    } catch {
      return false
    }
  }
}

import type { Dialog, Page } from 'playwright'
import {
  CpiResearchResultSaver,
  type CpiResearchSaveFrame,
  type CpiResearchSaveResult
} from './cpi-research-result-saver'

const EXPECTED_CONFIRMATION_PATTERN = /deseja\s+realmente\s+alterar\s+esta\s+pesquisa/i

export interface CpiResearchSavePort {
  saveOnce(frame: CpiResearchSaveFrame): Promise<CpiResearchSaveResult>
}

export interface CpiResearchConfirmedSaveOptions {
  dialogTimeoutMs?: number
}

export interface CpiResearchConfirmedSaveResult extends CpiResearchSaveResult {
  confirmationAccepted: true
  confirmationMessage: string
}

export class CpiResearchConfirmedSaveController {
  private readonly dialogTimeoutMs: number

  constructor(
    private readonly saver: CpiResearchSavePort = new CpiResearchResultSaver(),
    options: CpiResearchConfirmedSaveOptions = {}
  ) {
    this.dialogTimeoutMs = options.dialogTimeoutMs ?? 10_000

    if (this.dialogTimeoutMs < 1) {
      throw new Error('O tempo limite da confirmação deve ser maior que zero.')
    }
  }

  async saveOnce(page: Page, frame: CpiResearchSaveFrame): Promise<CpiResearchConfirmedSaveResult> {
    const dialogPromise = page.waitForEvent('dialog', {
      timeout: this.dialogTimeoutMs
    })

    const savePromise = this.saver.saveOnce(frame)

    let dialog: Dialog

    try {
      dialog = await dialogPromise
    } catch {
      await savePromise.catch(() => undefined)

      throw new Error('A confirmação da CPI não apareceu após clicar em Alterar.')
    }

    const confirmationMessage = this.sanitizeMessage(dialog.message())

    const isExpectedConfirmation =
      dialog.type() === 'confirm' && EXPECTED_CONFIRMATION_PATTERN.test(confirmationMessage)

    if (!isExpectedConfirmation) {
      await dialog.dismiss()
      await savePromise.catch(() => undefined)

      throw new Error('A CPI apresentou um diálogo diferente da confirmação esperada.')
    }

    await dialog.accept()

    const saveResult = await savePromise

    return {
      ...saveResult,
      confirmationAccepted: true,
      confirmationMessage
    }
  }

  private sanitizeMessage(value: string): string {
    return value
      .replace(/\b\d{3}\.?\d{3}\.?\d{3}-?\d{2}\b/g, '[DADO_REMOVIDO]')
      .replace(/\b\d{11}\b/g, '[DADO_REMOVIDO]')
      .replace(/\s+/g, ' ')
      .trim()
      .slice(0, 500)
  }
}

import { setTimeout as wait } from 'node:timers/promises'
import type { ResearchClassification } from '../../domain/research/research-classification'
import { mapResearchClassificationToCpiResult } from './cpi-research-result-mapper'

export interface CpiResearchResultLocator {
  count(): Promise<number>
  click?(): Promise<unknown>
  isChecked?(): Promise<boolean>
  selectOption?(option: { label: string }): Promise<unknown>
  innerText?(options?: { timeout?: number }): Promise<string>
}

export interface CpiResearchResultFrame {
  locator(selector: string): CpiResearchResultLocator
}

export interface CpiResearchResultControllerOptions {
  resultSearchAttempts?: number
  resultSearchIntervalMs?: number
  stabilizationAttempts?: number
}

export interface PreparedCpiResearchResult {
  classification: ResearchClassification
  selectedResultLabel: string
  researcherLabel: string
  primaryResearcherLabel: string
  statusLabel: string
  responseChanged: false
  saved: false
}

const RESEARCHER_SELECTOR = 'select[name="id_pesquisador"]'

const PRIMARY_RESEARCHER_SELECTOR = 'select[name="id_pesquisador_prim"]'

const STATUS_SELECTOR = 'select[name="id_pesq_status"]'

const RESEARCHER_LABEL = 'Desk Data'
const STATUS_LABEL = 'Respondida'

export class CpiResearchResultController {
  private readonly resultSearchAttempts: number
  private readonly resultSearchIntervalMs: number
  private readonly stabilizationAttempts: number

  constructor(options: CpiResearchResultControllerOptions = {}) {
    this.resultSearchAttempts = options.resultSearchAttempts ?? 40

    this.resultSearchIntervalMs = options.resultSearchIntervalMs ?? 250

    this.stabilizationAttempts = options.stabilizationAttempts ?? 3

    if (this.resultSearchAttempts < 1) {
      throw new Error('Número de tentativas deve ser maior que zero.')
    }

    if (this.resultSearchIntervalMs < 0) {
      throw new Error('Intervalo de busca não pode ser negativo.')
    }

    if (this.stabilizationAttempts < 1) {
      throw new Error('Número de estabilizações deve ser maior que zero.')
    }
  }

  async prepare(
    frame: CpiResearchResultFrame,
    classification: ResearchClassification
  ): Promise<PreparedCpiResearchResult> {
    const resultOption = mapResearchClassificationToCpiResult(classification)

    for (let attempt = 0; attempt < this.stabilizationAttempts; attempt += 1) {
      await this.selectByLabel(frame, STATUS_SELECTOR, STATUS_LABEL, 'Status')

      const resultRadio = await this.waitForSingleControl(
        frame,
        resultOption.selector,
        'Opção de resultado da CPI'
      )

      const resultId = resultOption.selector.replace(/^#/, '')

      const resultLabel = await this.waitForSingleControl(
        frame,
        'label[for="' + resultId + '"]',
        'Rótulo visível da opção de resultado da CPI'
      )

      await this.selectByLabel(frame, RESEARCHER_SELECTOR, RESEARCHER_LABEL, 'Pesquisador/Fonte')

      await this.selectByLabel(
        frame,
        PRIMARY_RESEARCHER_SELECTOR,
        RESEARCHER_LABEL,
        'Pesquisador/Fonte 1º'
      )

      if (!resultLabel.click) {
        throw new Error('O rótulo da opção de resultado não permite clique.')
      }

      await resultLabel.click()

      if (await this.isPreparedState(frame, resultRadio, resultOption.label)) {
        return {
          classification,
          selectedResultLabel: resultOption.label,
          researcherLabel: RESEARCHER_LABEL,
          primaryResearcherLabel: RESEARCHER_LABEL,
          statusLabel: STATUS_LABEL,
          responseChanged: false,
          saved: false
        }
      }

      if (attempt < this.stabilizationAttempts - 1) {
        await wait(this.resultSearchIntervalMs)
      }
    }

    throw new Error('Os campos da CPI não permaneceram estáveis. ' + 'A gravação foi bloqueada.')
  }

  private async selectByLabel(
    frame: CpiResearchResultFrame,
    selector: string,
    label: string,
    fieldName: string
  ): Promise<void> {
    const control = await this.waitForSingleControl(frame, selector, fieldName)

    if (!control.selectOption) {
      throw new Error(fieldName + ' não permite seleção.')
    }

    await control.selectOption({ label })
  }

  private async isPreparedState(
    frame: CpiResearchResultFrame,
    resultRadio: CpiResearchResultLocator,
    expectedResultLabel: string
  ): Promise<boolean> {
    const [statusOk, researcherOk, primaryResearcherOk] = await Promise.all([
      this.hasSelectedLabel(frame, STATUS_SELECTOR, STATUS_LABEL, 'Status'),
      this.hasSelectedLabel(frame, RESEARCHER_SELECTOR, RESEARCHER_LABEL, 'Pesquisador/Fonte'),
      this.hasSelectedLabel(
        frame,
        PRIMARY_RESEARCHER_SELECTOR,
        RESEARCHER_LABEL,
        'Pesquisador/Fonte 1º'
      )
    ])

    if (!resultRadio.isChecked) {
      throw new Error('Não foi possível confirmar a opção de resultado.')
    }

    const resultOk = await resultRadio.isChecked()

    return (
      statusOk && researcherOk && primaryResearcherOk && resultOk && expectedResultLabel.length > 0
    )
  }

  private async hasSelectedLabel(
    frame: CpiResearchResultFrame,
    selector: string,
    expectedLabel: string,
    fieldName: string
  ): Promise<boolean> {
    const selectedOption = frame.locator(selector + ' option:checked')

    const selectedCount = await selectedOption.count().catch(() => 0)

    if (selectedCount === 0) {
      return false
    }

    if (selectedCount !== 1) {
      throw new Error(
        fieldName +
          ' deveria possuir exatamente uma opção selecionada. Encontradas: ' +
          selectedCount
      )
    }

    if (!selectedOption.innerText) {
      throw new Error('Não foi possível ler a opção selecionada de ' + fieldName + '.')
    }

    const selectedLabel = this.normalizeLabel(
      await selectedOption.innerText({
        timeout: 5_000
      })
    )

    return selectedLabel === this.normalizeLabel(expectedLabel)
  }

  private normalizeLabel(value: string): string {
    return value.replace(/\*/g, '').replace(/\s+/g, ' ').trim()
  }

  private async waitForSingleControl(
    frame: CpiResearchResultFrame,
    selector: string,
    fieldName: string
  ): Promise<CpiResearchResultLocator> {
    const control = frame.locator(selector)

    for (let attempt = 0; attempt < this.resultSearchAttempts; attempt += 1) {
      const count = await control.count()

      if (count === 1) {
        return control
      }

      if (count > 1) {
        throw new Error(
          fieldName + ' deveria possuir exatamente um controle. Encontrados: ' + count
        )
      }

      if (attempt < this.resultSearchAttempts - 1) {
        await wait(this.resultSearchIntervalMs)
      }
    }

    throw new Error(fieldName + ' não apareceu após aguardar a renderização.')
  }

export type CpiQueueAvailabilityState = (typeof CPI_QUEUE_AVAILABILITY_STATES)[number]

function normalizeText(value: string): string {
  return value
    .normalize('NFD')
    .replace(/\p{Diacritic}/gu, '')
    .replace(/\s+/g, ' ')
    .trim()
}

export function parseCpiQueueTotal(pageText: string): number | null {
  const normalized = normalizeText(pageText)

  const match = normalized.match(
    /Mostrando:\s*\d+\s*a\s*\d+\s*\(\s*de\s*(\d+)\s*registro(?:s)?\s*\)/i
  )

  if (!match) {
    return null
  }

  const total = Number(match[1])

  return Number.isInteger(total) && total >= 0 ? total : null
}

export function classifyCpiQueueText(pageText: string): CpiQueueAvailabilityState {
  const total = parseCpiQueueTotal(pageText)

  if (total === null) {
    return 'INDETERMINATE'
  }

  return total === 0 ? 'EMPTY' : 'HAS_RESEARCH'
}

export class CpiQueueAvailabilityReader {
  async read(page: Page): Promise<CpiQueueAvailabilityState> {
    for (const frame of page.frames()) {
      const bodyText = await frame
        .locator('body')
        .innerText({
          timeout: 5_000
        })
        .catch(() => '')

      const state = classifyCpiQueueText(bodyText)

      if (state !== 'INDETERMINATE') {
        return state
      }
    }

    return 'INDETERMINATE'
  }
}

import { setTimeout as wait } from 'node:timers/promises'
import type { Page } from 'playwright'
import type { CpiQueueAvailabilityState } from './cpi-queue-availability'

export interface CpiQueueRefresher {
  refreshQueue(page: Page): Promise<void>
}

export interface CpiQueueAvailabilitySource {
  read(page: Page): Promise<CpiQueueAvailabilityState>
}

export interface CpiQueueDelay {
  wait(milliseconds: number, signal?: AbortSignal): Promise<void>
}

export interface CpiQueueMonitorOptions {
  isPauseRequested?: () => boolean
  signal?: AbortSignal
  onState?: (state: CpiQueueAvailabilityState, checks: number) => void
}

export interface CpiQueueMonitorResult {
  status: 'RESEARCH_AVAILABLE' | 'PAUSED'
  checks: number
}

class NodeQueueDelay implements CpiQueueDelay {
  async wait(milliseconds: number, signal?: AbortSignal): Promise<void> {
    await wait(milliseconds, undefined, {
      signal
    })
  }
}

export class CpiQueueMonitor {
  constructor(
    private readonly refresher: CpiQueueRefresher,
    private readonly availabilitySource: CpiQueueAvailabilitySource,
    private readonly delay: CpiQueueDelay = new NodeQueueDelay(),
    private readonly intervalMs = 60_000
  ) {
    if (intervalMs < 30_000) {
      throw new Error('Intervalo da fila deve ser de pelo menos 30000 ms')
    }
  }

  async waitForResearch(
    page: Page,
    options: CpiQueueMonitorOptions = {}
  ): Promise<CpiQueueMonitorResult> {
    let checks = 0

    while (true) {
      if (options.signal?.aborted || options.isPauseRequested?.()) {
        return {
          status: 'PAUSED',
          checks
        }
      }

      await this.refresher.refreshQueue(page)

      checks += 1

      const state = await this.availabilitySource.read(page)

      options.onState?.(state, checks)

      if (state === 'HAS_RESEARCH') {
        return {
          status: 'RESEARCH_AVAILABLE',
          checks
        }
      }

      if (state === 'INDETERMINATE') {
        throw new Error('Não foi possível determinar o estado da fila da CPI')
      }

      if (options.signal?.aborted || options.isPauseRequested?.()) {
        return {
          status: 'PAUSED',
          checks
        }
      }

      try {
        await this.delay.wait(this.intervalMs, options.signal)
      } catch (error: unknown) {
        if (options.signal?.aborted) {
          return {
            status: 'PAUSED',
            checks
          }
        }

        throw error
      }
    }
  }
}

  }

  async applyDatasetIds(
    page: DeskDataDatasetPage,
    requestedIds: readonly string[]
  ): Promise<DeskDataDatasetSnapshot> {
    const selectedIds = [...new Set(requestedIds.map((id) => id.trim()).filter(Boolean))]

    if (selectedIds.length === 0) {
      throw new Error('Informe pelo menos um conjunto de dados da DeskData')
    }

    const checkboxes = page.locator(DESKDATA_DATASET_SELECTORS.allCheckboxes)

    const checkboxCount = await checkboxes.count()

    if (checkboxCount === 0) {
      throw new Error('Nenhum conjunto de dados foi encontrado na DeskData')
    }

    const selectedIdSet = new Set(selectedIds)
    const foundIds = new Set<string>()

    for (let index = 0; index < checkboxCount; index += 1) {
      const checkbox = checkboxes.nth(index)
      const checkboxId = await checkbox.getAttribute('id')

      if (!checkboxId) {
        continue
      }

      const shouldBeChecked = selectedIdSet.has(checkboxId)

      if (shouldBeChecked) {
        foundIds.add(checkboxId)
      }

      await checkbox.evaluate((element, targetState) => {
        const input = element as HTMLInputElement

        if (input.checked === targetState) {
          return
        }

        input.checked = targetState

        input.dispatchEvent(
          new Event('input', {
            bubbles: true
          })
        )

        input.dispatchEvent(
          new Event('change', {
            bubbles: true
          })
        )
      }, shouldBeChecked)

      if ((await checkbox.isChecked()) !== shouldBeChecked) {
        throw new Error(`A DeskData não manteve o estado do conjunto ${checkboxId}`)
      }
    }

    const missingIds = selectedIds.filter((id) => !foundIds.has(id))

    if (missingIds.length > 0) {
      throw new Error(
        `Conjuntos obrigatórios não encontrados na DeskData: ${missingIds.join(', ')}`
      )
    }

    return {
      selectedIds,
      selectedCount: selectedIds.length
    }
  }

    throw new Error('O resultado da DeskData não ficou disponível dentro do tempo limite.')
  }

  private extractPath(pageUrl: string): string {
    try {
      return new URL(pageUrl).pathname

import { randomBytes } from 'node:crypto'
import { mkdir, open, rm, stat } from 'node:fs/promises'
import { extname, join } from 'node:path'
import type { BrowserContext, Download, Page } from 'playwright'

export const DESKDATA_PDF_EXPORT_SELECTOR =
  'form[method="POST"] input[type="submit"][name="export"][value="Exportar"][formtarget="_blank"]'

const DEFAULT_OUTPUT_DIRECTORY = 'E:\\PDF-AGENTE-CPI'

type DownloadState = 'IDLE' | 'DOWNLOADING' | 'DOWNLOADED'

export interface DeskDataPdfDownloadControllerOptions {
  outputDirectory?: string
  downloadTimeoutMs?: number
}

export interface DeskDataPdfDownloadResult {
  filePath: string
  sizeBytes: number
}

export class DeskDataPdfDownloadController {
  private readonly outputDirectory: string
  private readonly downloadTimeoutMs: number
  private state: DownloadState = 'IDLE'

  constructor(options: DeskDataPdfDownloadControllerOptions = {}) {
    this.outputDirectory = options.outputDirectory ?? DEFAULT_OUTPUT_DIRECTORY
    this.downloadTimeoutMs = options.downloadTimeoutMs ?? 30_000

    if (this.downloadTimeoutMs < 1) {
      throw new Error('O tempo limite do download deve ser maior que zero.')
    }
  }

  async downloadOnce(
    context: BrowserContext,
    sourcePage: Page
  ): Promise<DeskDataPdfDownloadResult> {
    if (this.state !== 'IDLE') {
      throw new Error('O PDF da DeskData já foi exportado ou está sendo exportado nesta execução.')
    }

    const exportControl = sourcePage.locator(DESKDATA_PDF_EXPORT_SELECTOR)

    const exportControlCount = await exportControl.count()

    if (exportControlCount !== 1) {
      throw new Error(
        'O botão Exportar deveria possuir exatamente um controle. Encontrados: ' +
          exportControlCount
      )
    }

    this.state = 'DOWNLOADING'

    let savedFilePath: string | undefined
    let timeoutHandle: ReturnType<typeof setTimeout> | undefined

    let resolveDownload: (download: Download) => void

    const downloadPromise = new Promise<Download>((resolve) => {
      resolveDownload = resolve
    })

    let downloadCaptured = false

    const onDownload = (download: Download): void => {
      if (downloadCaptured) {
        return
      }

      downloadCaptured = true
      resolveDownload(download)
    }

    const registeredPages = new Set<Page>()

    const registerPage = (page: Page): void => {
      if (registeredPages.has(page)) {
        return
      }

      registeredPages.add(page)
      page.on('download', onDownload)
    }

    const onPage = (page: Page): void => {
      registerPage(page)
    }

    context.pages().forEach(registerPage)
    registerPage(sourcePage)
    context.on('page', onPage)

    try {
      const timeoutPromise = new Promise<never>((_resolve, reject) => {
        timeoutHandle = setTimeout(() => {
          reject(new Error('O download do PDF da DeskData não iniciou dentro do tempo limite.'))
        }, this.downloadTimeoutMs)
      })

      await exportControl.click({
        timeout: 10_000
      })

      const download = await Promise.race([downloadPromise, timeoutPromise])

      const downloadFailure = await download.failure()

      if (downloadFailure) {
        throw new Error('A DeskData informou falha no download do PDF: ' + downloadFailure)
      }

      await mkdir(this.outputDirectory, {
        recursive: true
      })

      savedFilePath = join(this.outputDirectory, this.createTechnicalFileName())

      await download.saveAs(savedFilePath)

      const sizeBytes = await this.validatePdf(savedFilePath)

      this.state = 'DOWNLOADED'

      return {
        filePath: savedFilePath,
        sizeBytes
      }
    } catch (error: unknown) {
      this.state = 'IDLE'

      if (savedFilePath) {
        await rm(savedFilePath, {
          force: true
        })
      }

      throw error
    } finally {
      if (timeoutHandle) {
        clearTimeout(timeoutHandle)
      }

      context.off('page', onPage)

      registeredPages.forEach((page) => {
        page.off('download', onDownload)
      })
    }
  }

  private createTechnicalFileName(): string {
    const timestamp = new Date()
      .toISOString()
      .replace(/\.\d{3}Z$/, '')
      .replace(/[-:]/g, '')
      .replace('T', '-')

    const token = randomBytes(4).toString('hex')

    return 'deskdata-' + timestamp + '-' + token + '.pdf'
  }

  private async validatePdf(filePath: string): Promise<number> {
    if (extname(filePath).toLowerCase() !== '.pdf') {
      throw new Error('O arquivo baixado não possui extensão PDF.')
    }

    const fileStat = await stat(filePath)

    if (!fileStat.isFile() || fileStat.size < 5) {
      throw new Error('O arquivo PDF baixado está vazio ou inválido.')
    }

    const signature = Buffer.alloc(5)
    const fileHandle = await open(filePath, 'r')

    try {
      const readResult = await fileHandle.read(signature, 0, signature.length, 0)

      if (readResult.bytesRead !== signature.length || signature.toString('ascii') !== '%PDF-') {
        throw new Error('O arquivo baixado não possui uma assinatura PDF válida.')
      }
    } finally {
      await fileHandle.close()
    }

    return fileStat.size
  }
}
