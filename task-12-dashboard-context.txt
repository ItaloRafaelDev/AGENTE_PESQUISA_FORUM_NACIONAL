import { join } from 'node:path'
import { createInterface } from 'node:readline/promises'
import { stdin, stdout } from 'node:process'
import { chromium, type BrowserContext, type Page } from 'playwright'
import { PlatformPageManager } from '../src/automation/browser/platform-page-manager'
import { CpiQueueFilterController } from '../src/automation/cpi/cpi-queue-filter-controller'
import { ContinuousResearchStageError } from '../src/automation/flow/continuous-research-stage'
import {
  ContinuousResearchRunner,
  type ContinuousResearchRunnerState
} from '../src/automation/flow/continuous-research-runner'
import { SafeResearchLogger } from '../src/automation/logging/safe-research-logger'

const CPI_QUEUE_PATH = '/zpersespecial/pesquisas/aguardando/mod_t_aguardando.aspx'

type TerminalInterface = ReturnType<typeof createInterface>

interface AuthenticatedPages {
  cpiPage: Page
  deskDataPage: Page
}

function findPage(pages: Page[], hostname: string): Page | undefined {
  return [...pages].reverse().find((page) => {
    if (page.isClosed()) {
      return false
    }

    try {
      const currentHostname = new URL(page.url()).hostname

      return currentHostname === hostname || currentHostname.endsWith('.' + hostname)
    } catch {
      return false
    }
  })
}

function isCpiQueuePage(page: Page): boolean {
  try {
    return new URL(page.url()).pathname.toLowerCase() === CPI_QUEUE_PATH
  } catch {
    return false
  }
}

function isDeskDataPeoplePage(page: Page): boolean {
  try {
    const url = new URL(page.url())

    const validHostname =
      url.hostname === 'deskdata.com.br' || url.hostname.endsWith('.deskdata.com.br')

    const validPath = url.pathname === '/pessoas' || url.pathname.startsWith('/pessoas/')

    return validHostname && validPath
  } catch {
    return false
  }
}

async function hasVisiblePasswordField(page: Page): Promise<boolean> {
  return (
    (await page
      .locator('input[type="password"]:visible')
      .count()
      .catch(() => 0)) > 0
  )
}

async function ensureAuthenticatedPages(
  context: BrowserContext,
  terminal: TerminalInterface,
  logger: SafeResearchLogger
): Promise<AuthenticatedPages> {
  await new PlatformPageManager().ensurePages(context)

  let cpiPage = findPage(context.pages(), 'adm.cpicentral.com.br')

  let deskDataPage = findPage(context.pages(), 'deskdata.com.br')

  if (!cpiPage || !deskDataPage) {
    throw new Error('As abas da CPI e da DeskData não foram localizadas.')
  }

  if (!isCpiQueuePage(cpiPage)) {
    logger.info('Aguardando login manual na CPI.')

    await terminal.question('Faça o login manual na CPI, deixe a fila aberta e pressione ENTER...')

    cpiPage = findPage(context.pages(), 'adm.cpicentral.com.br')
  }

  if (!cpiPage || !isCpiQueuePage(cpiPage)) {
    throw new Error('A fila autenticada da CPI não foi confirmada.')
  }

  const deskDataRequiresLogin =
    !isDeskDataPeoplePage(deskDataPage) || (await hasVisiblePasswordField(deskDataPage))

  if (deskDataRequiresLogin) {
    logger.info('Aguardando login manual na DeskData.')

    await terminal.question('Faça o login manual na DeskData e pressione ENTER...')

    deskDataPage = findPage(context.pages(), 'deskdata.com.br')
  }

  if (
    !deskDataPage ||
    !isDeskDataPeoplePage(deskDataPage) ||
    (await hasVisiblePasswordField(deskDataPage))
  ) {
    throw new Error('A sessão autenticada da DeskData não foi confirmada.')
  }

  return {
    cpiPage,
    deskDataPage
  }
}

function logRunnerState(logger: SafeResearchLogger, state: ContinuousResearchRunnerState): void {
  switch (state.type) {
    case 'WAITING_QUEUE':
      if (state.checks === 0) {
        logger.info('Verificando a fila da CPI.')
      } else {
        logger.info('Verificação da fila concluída: ' + state.checks)
      }

      return

    case 'IDLE':
      logger.info('Fila vazia. Nova verificação em ' + state.nextCheckInMs / 1000 + ' segundos.')

      return

    case 'RESEARCH_AVAILABLE':
      logger.info('Pesquisa disponível. Iniciando ciclo ' + state.cycle + '.')

      return

    case 'CYCLE_COMPLETED':
      logger.info('Ciclo ' + state.cycle + ' concluído e gravado.')

      return

    case 'STOPPED':
      logger.info('Execução contínua interrompida. Ciclos concluídos: ' + state.cycles + '.')

      return
  }
}

async function main(): Promise<void> {
  const appData = process.env.APPDATA

  if (!appData) {
    throw new Error('Variável APPDATA não encontrada.')
  }

  const logger = new SafeResearchLogger()

  const terminal = createInterface({
    input: stdin,
    output: stdout
  })

  const abortController = new AbortController()

  let sigintReceived = false
  let sigintRegistered = false

  const handleSigint = (): void => {
    if (sigintReceived) {
      return
    }

    sigintReceived = true

    logger.info('Interrupção solicitada. O agente será encerrado no próximo ponto seguro.')

    abortController.abort()
  }

  const context = await chromium.launchPersistentContext(
    join(appData, 'AgenteCPI', 'browser-profile'),
    {
      headless: false,
      viewport: null,
      acceptDownloads: true,
      args: ['--start-maximized']
    }
  )

  try {
    const { cpiPage, deskDataPage } = await ensureAuthenticatedPages(context, terminal, logger)

    await cpiPage.bringToFront()

    logger.info('Aplicando filtro 29 na CPI.')

    await new CpiQueueFilterController().applyCriminalForumFilter(cpiPage)

    logger.info('Modo automático contínuo iniciado.')

    logger.info('Pressione Ctrl+C para solicitar encerramento seguro.')

    process.once('SIGINT', handleSigint)

    sigintRegistered = true

    const result = await new ContinuousResearchRunner().run({
      context,
      cpiPage,
      deskDataPage,
      signal: abortController.signal,
      onState: (state): void => {
        logRunnerState(logger, state)
      }
    })

    logger.info('Agente finalizado. Ciclos concluídos: ' + result.completedCycles + '.')
  } catch (error: unknown) {
    const stage = error instanceof ContinuousResearchStageError ? error.stage : 'CONTINUOUS_RUNNER'

    logger.error(stage, error)

    logger.info('Política E1 aplicada. Nenhuma nova pesquisa será iniciada.')

    process.exitCode = 1

    await terminal.question(
      'O navegador permanecerá aberto para análise. Pressione ENTER para encerrar...'
    )
  } finally {
    if (sigintRegistered) {
      process.off('SIGINT', handleSigint)
    }

    terminal.close()

    await context.close()

    logger.info('Navegador encerrado com segurança.')
  }
}

import { setTimeout as wait } from 'node:timers/promises'
import type { Page } from 'playwright'
import type { CpiQueueAvailabilityState } from './cpi-queue-availability'

export interface CpiQueueRefresher {
  refreshQueue(page: Page): Promise<void>
}

export interface CpiQueueAvailabilitySource {
  read(page: Page): Promise<CpiQueueAvailabilityState>
}

export interface CpiQueueDelay {
  wait(milliseconds: number, signal?: AbortSignal): Promise<void>
}

export interface CpiQueueMonitorOptions {
  isPauseRequested?: () => boolean
  signal?: AbortSignal
  onState?: (state: CpiQueueAvailabilityState, checks: number) => void
}

export interface CpiQueueMonitorResult {
  status: 'RESEARCH_AVAILABLE' | 'PAUSED'
  checks: number
}

class NodeQueueDelay implements CpiQueueDelay {
  async wait(milliseconds: number, signal?: AbortSignal): Promise<void> {
    await wait(milliseconds, undefined, {
      signal
    })
  }
}

export class CpiQueueMonitor {
  constructor(
    private readonly refresher: CpiQueueRefresher,
    private readonly availabilitySource: CpiQueueAvailabilitySource,
    private readonly delay: CpiQueueDelay = new NodeQueueDelay(),
    private readonly intervalMs = 60_000
  ) {
    if (intervalMs < 30_000) {
      throw new Error('Intervalo da fila deve ser de pelo menos 30000 ms')
    }
  }

  async waitForResearch(
    page: Page,
    options: CpiQueueMonitorOptions = {}
  ): Promise<CpiQueueMonitorResult> {
    let checks = 0

    while (true) {
      if (options.signal?.aborted || options.isPauseRequested?.()) {
        return {
          status: 'PAUSED',
          checks
        }
      }

      await this.refresher.refreshQueue(page)

      checks += 1

      const state = await this.availabilitySource.read(page)

      options.onState?.(state, checks)

      if (state === 'HAS_RESEARCH') {
        return {
          status: 'RESEARCH_AVAILABLE',
          checks
        }
      }

      if (state === 'INDETERMINATE') {
        throw new Error('Não foi possível determinar o estado da fila da CPI')
      }

      if (options.signal?.aborted || options.isPauseRequested?.()) {
        return {
          status: 'PAUSED',
          checks
        }
      }

      try {
        await this.delay.wait(this.intervalMs, options.signal)
      } catch (error: unknown) {
        if (options.signal?.aborted) {
          return {
            status: 'PAUSED',
            checks
          }
        }

        throw error
      }
    }
  }
}

import type { BrowserContext, Page } from 'playwright'
import type { CpiQueueAvailabilityState } from '../cpi/cpi-queue-availability'
import { CpiQueueAvailabilityReader } from '../cpi/cpi-queue-availability'
import { CpiQueueFilterController } from '../cpi/cpi-queue-filter-controller'

  type CpiQueueMonitorResult

  SingleResearchCycle,
  type SingleResearchCycleInput,
  type SingleResearchCycleResult
} from './single-research-cycle'

export const CONTINUOUS_QUEUE_INTERVAL_MS = 60_000 as const

export interface ContinuousResearchRunnerInput {
  context: BrowserContext
  cpiPage: Page
  deskDataPage: Page
  signal: AbortSignal
  onState?: (state: ContinuousResearchRunnerState) => void
}

export type ContinuousResearchRunnerState =
  | {
      type: 'WAITING_QUEUE'
      checks: number
    }
  | {
      type: 'IDLE'
      nextCheckInMs: typeof CONTINUOUS_QUEUE_INTERVAL_MS
    }
  | {
      type: 'RESEARCH_AVAILABLE'
      cycle: number
      checks: number
    }
  | {
      type: 'CYCLE_COMPLETED'
      cycle: number
    }
  | {
      type: 'STOPPED'
      cycles: number
    }

export interface ContinuousResearchRunResult {
  status: 'STOPPED_BY_SIGNAL'
  completedCycles: number
}

export interface ContinuousQueueMonitorPort {
  waitForResearch(page: Page, options?: CpiQueueMonitorOptions): Promise<CpiQueueMonitorResult>
}

export interface SingleResearchCyclePort {
  run(input: SingleResearchCycleInput): Promise<SingleResearchCycleResult>
}

export type SingleResearchCycleFactory = () => SingleResearchCyclePort

export interface ContinuousResearchRunnerDependencies {
  queueMonitor: ContinuousQueueMonitorPort
  cycleFactory: SingleResearchCycleFactory
}

function createDefaultDependencies(): ContinuousResearchRunnerDependencies {
  const filterController = new CpiQueueFilterController()

  return {
    queueMonitor: new CpiQueueMonitor(filterController, new CpiQueueAvailabilityReader()),
    cycleFactory: () => new SingleResearchCycle()
  }
}

export class ContinuousResearchRunner {
  constructor(
    private readonly dependencies: ContinuousResearchRunnerDependencies = createDefaultDependencies()
  ) {}

  async run(input: ContinuousResearchRunnerInput): Promise<ContinuousResearchRunResult> {
    let completedCycles = 0

    while (!input.signal.aborted) {
      input.onState?.({
        type: 'WAITING_QUEUE',
        checks: 0
      })

      const queueResult = await this.dependencies.queueMonitor.waitForResearch(input.cpiPage, {
        signal: input.signal,
        onState: (state, checks) => {
          this.reportQueueState(input, state, checks)
        }
      })

      if (queueResult.status === 'PAUSED') {
        break
      }

      const cycleNumber = completedCycles + 1

      input.onState?.({
        type: 'RESEARCH_AVAILABLE',
        cycle: cycleNumber,
        checks: queueResult.checks
      })

      const cycle = this.dependencies.cycleFactory()

      await cycle.run({
        context: input.context,
        cpiPage: input.cpiPage,
        deskDataPage: input.deskDataPage,
        signal: input.signal
      })

      completedCycles = cycleNumber

      input.onState?.({
        type: 'CYCLE_COMPLETED',
        cycle: completedCycles
      })
    }

    input.onState?.({
      type: 'STOPPED',
      cycles: completedCycles
    })

    return {
      status: 'STOPPED_BY_SIGNAL',
      completedCycles
    }
  }

  private reportQueueState(
    input: ContinuousResearchRunnerInput,
    state: CpiQueueAvailabilityState,
    checks: number
  ): void {
    input.onState?.({
      type: 'WAITING_QUEUE',
      checks
    })

    if (state !== 'EMPTY') {
      return
    }

    input.onState?.({
      type: 'IDLE',
      nextCheckInMs: CONTINUOUS_QUEUE_INTERVAL_MS
    })
  }
}

    saveController: new CpiResearchConfirmedSaveController(),
    queueReturnController: new DefaultCpiQueueReturnController()
  }
}

export class SingleResearchCycle {
  constructor(
    private readonly dependencies: SingleResearchCycleDependencies = createDefaultDependencies()
  ) {}

  async run(input: SingleResearchCycleInput): Promise<SingleResearchCycleResult> {
    this.assertCanStart(input.signal)

    await this.runStage('OPENING_CPI_RESEARCH', async () => {
      await this.dependencies.opener.openFirst(input.cpiPage)
    })

    const identity = await this.runStage('READING_CPI_IDENTITY', () =>
      this.dependencies.identityReader.read(input.cpiPage)
    )

    const detailFrame = await this.runStage('READING_CPI_IDENTITY', async () =>
      this.findDetailFrame(input.cpiPage, identity.detailFramePath)
    )

    await this.runStage('PREPARING_DESKDATA', () =>
      this.dependencies.nextSearchController.prepare(input.deskDataPage)
    )

    const preparation = await this.runStage('PREPARING_CPI_RESULT', () =>
      this.dependencies.preparationWorkflow.prepare(
        input.context,
        input.deskDataPage,
        detailFrame,
        identity
      )
    )

    this.validatePreparation(preparation)

    const saveResult = await this.runStage('SAVING_CPI', () =>
      this.dependencies.saveController.saveOnce(input.cpiPage, preparation.detailFrame)
    )

    if (!saveResult.saved) {
      throw new ContinuousResearchStageError(
        'SAVING_CPI',
        new Error('A gravação da pesquisa não foi confirmada.')
      )
    }

    await this.runStage('PREPARING_NEXT_SEARCH', () =>
      this.dependencies.nextSearchController.prepare(input.deskDataPage)
    )

    await this.runStage('RETURNING_TO_QUEUE', () =>
      this.dependencies.queueReturnController.returnToQueue(input.cpiPage)
    )

    return {
      maskedCpf: preparation.maskedCpf,
      selectedDatasetCount: preparation.selectedDatasetCount,
      classification: preparation.classification.classification,
      pdfPath: preparation.pdf.filePath,
      pdfSizeBytes: preparation.pdf.sizeBytes,
      saved: true,
      confirmationAccepted: saveResult.confirmationAccepted
    }
  }

  private assertCanStart(signal: AbortSignal | undefined): void {
    if (!signal?.aborted) {
      return
    }

    throw new ContinuousResearchStageError(
      'OPENING_CPI_RESEARCH',
      new Error('A execução foi interrompida antes do início da pesquisa.')
    )
  }

  private findDetailFrame(page: Page, expectedPath: string): CpiResearchPdfPreparationFrame {
    const normalizedExpectedPath = expectedPath.toLowerCase()

    const detailFrames = page.frames().filter((frame: Frame) => {
      try {
        return new URL(frame.url()).pathname.toLowerCase() === normalizedExpectedPath
      } catch {
        return false
      }
    })

    if (detailFrames.length !== 1) {
      throw new Error(
        'O formulário interno da pesquisa deveria possuir exatamente um frame. Encontrados: ' +
          detailFrames.length
      )
    }

    return detailFrames[0] as CpiResearchPdfPreparationFrame
  }

  private validatePreparation(preparation: FullResearchPreparationResult): void {
    if (!preparation.cpi.attachment.attached) {
      throw new ContinuousResearchStageError(
        'ATTACHING_PDF',
        new Error('O PDF não foi confirmado no campo de anexo da CPI.')
      )
    }

    if (preparation.pdf.sizeBytes < 1) {
      throw new ContinuousResearchStageError(
        'DOWNLOADING_PDF',
        new Error('O PDF da DeskData possui tamanho inválido.')
      )
    }

    if (
      preparation.cpi.prepared.researcherLabel !== 'Desk Data' ||
      preparation.cpi.prepared.primaryResearcherLabel !== 'Desk Data' ||
      preparation.cpi.prepared.statusLabel !== 'Respondida'
    ) {
      throw new ContinuousResearchStageError(
        'PREPARING_CPI_RESULT',
        new Error('Os campos obrigatórios da CPI não foram confirmados.')
      )
    }
  }

  private async runStage<T>(
    stage: ContinuousResearchStage,
    operation: () => Promise<T>
  ): Promise<T> {
    try {
      return await operation()
    } catch (error: unknown) {
      if (error instanceof ContinuousResearchStageError) {
        throw error
      }

      throw new ContinuousResearchStageError(stage, error)
    }
  }
}

const CPF_PATTERN = /\b\d{3}\.?\d{3}\.?\d{3}-?\d{2}\b/g

export type SafeLogWriter = (message: string) => void

export class SafeResearchLogger {
  constructor(
    private readonly write: SafeLogWriter = (message) => {
      process.stdout.write(message + '\n')
    }
  ) {}

  info(message: string): void {
    this.write(this.sanitizeMessage(message))
  }

  error(stage: string, error: unknown): void {
    const message = error instanceof Error ? error.message : String(error)

    this.write(this.sanitizeMessage('[' + stage + '] ' + message))
  }

  maskCpf(rawCpf: string): string {
    const digits = rawCpf.replace(/\D/g, '')

    if (digits.length !== 11) {
      return '[CPF_INVALIDO]'
    }

    return '***.***.***-' + digits.slice(-2)
  }

  sanitizeMessage(message: string): string {
    return message
      .replace(CPF_PATTERN, (cpf) => this.maskCpf(cpf))
      .replace(/\s+/g, ' ')
      .trim()
      .slice(0, 1000)
  }
}

import { useEffect, useState } from 'react'
import type { AgentControlSnapshot } from '../../../shared/types/agent-control-api'
import type { BrowserManagerSnapshot } from '../../../shared/types/browser-api'
import type {
  PlatformSessionsSnapshot,
  PlatformSessionState
} from '../../../shared/types/session-api'

type ActionName =
  'browser-start' | 'browser-stop' | 'agent-start' | 'agent-pause' | 'agent-resume' | 'agent-stop'

interface LogEntry {
  id: number
  message: string
  type: 'info' | 'success' | 'error'
  timestamp: string
}

const initialBrowserSnapshot: BrowserManagerSnapshot = {
  state: 'STOPPED',
  profilePath: '',
  startedAt: null
}

const initialAgentSnapshot: AgentControlSnapshot = {
  state: 'PARADO',
  pauseRequested: false,
  consecutiveErrors: 0,
  circuitOpen: false,
  safeMode: true
}

const initialSessionsSnapshot: PlatformSessionsSnapshot = {
  cpi: {
    state: 'INDETERMINATE',
    url: null
  },
  deskData: {
    state: 'INDETERMINATE',
    url: null
  },
  allAuthenticated: false,
  checkedAt: null
}

function formatState(state: string): string {
  return state.replaceAll('_', ' ')
}

function formatSessionState(state: PlatformSessionState): string {
  const labels: Record<PlatformSessionState, string> = {
    AUTHENTICATED: 'Autenticado',
    WAITING_LOGIN: 'Aguardando login',
    INDETERMINATE: 'Indeterminado'
  }

  return labels[state]
}

function formatTime(date: Date): string {
  return date.toLocaleTimeString('pt-BR', {
    hour: '2-digit',
    minute: '2-digit',
    second: '2-digit'
  })
}

function StatusIndicator({ active, label }: { active: boolean; label: string }): React.JSX.Element {
  return (
    <span className={active ? 'status-indicator active' : 'status-indicator'}>
      <span className="status-dot" />
      {label}
    </span>
  )
}

export default function AgentDashboard(): React.JSX.Element {
  const [browser, setBrowser] = useState(initialBrowserSnapshot)

  const [agent, setAgent] = useState(initialAgentSnapshot)

  const [sessions, setSessions] = useState(initialSessionsSnapshot)

  const [busyAction, setBusyAction] = useState<ActionName | null>(null)

  const [logs, setLogs] = useState<LogEntry[]>([])

  function addLog(message: string, type: LogEntry['type'] = 'info'): void {
    setLogs((currentLogs) =>
      [
        {
          id: Date.now(),
          message,
          type,
          timestamp: formatTime(new Date())
        },
        ...currentLogs
      ].slice(0, 10)
    )
  }

  useEffect(() => {
    let mounted = true

    async function loadInitialState(): Promise<void> {
      try {
        const [browserSnapshot, agentSnapshot, sessionSnapshot] = await Promise.all([
          window.agent.browser.getStatus(),
          window.agent.control.getStatus(),
          window.agent.sessions.getStatus()
        ])

        if (!mounted) {
          return
        }

        setBrowser(browserSnapshot)
        setAgent(agentSnapshot)
        setSessions(sessionSnapshot)
        addLog('Estado inicial carregado.')
      } catch (error: unknown) {
        if (!mounted) {
          return
        }

        addLog(
          error instanceof Error ? error.message : 'Falha ao carregar o estado inicial.',
          'error'
        )
      }
    }

    void loadInitialState()

    const operationalInterval = window.setInterval(() => {
      void Promise.all([window.agent.control.getStatus(), window.agent.sessions.getStatus()])
        .then(([agentSnapshot, sessionSnapshot]) => {
          if (!mounted) {
            return
          }

          setAgent(agentSnapshot)
          setSessions(sessionSnapshot)
        })
        .catch(() => undefined)
    }, 3_000)

    return () => {
      mounted = false

      window.clearInterval(operationalInterval)
    }
  }, [])

  async function executeAction<T>(
    actionName: ActionName,
    action: () => Promise<T>,
    applyResult: (result: T) => void,
    successMessage: string
  ): Promise<void> {
    setBusyAction(actionName)

    try {
      const result = await action()

      applyResult(result)
      addLog(successMessage, 'success')
    } catch (error: unknown) {
      addLog(error instanceof Error ? error.message : 'Falha inesperada.', 'error')
    } finally {
      setBusyAction(null)
    }
  }

  const browserRunning = browser.state === 'RUNNING'

  const agentStopped = agent.state === 'PARADO'

  const agentPaused = agent.state === 'PAUSADO'

  const operationBusy = busyAction !== null

  const canStartAgent = browserRunning && agentStopped && sessions.allAuthenticated

  return (
    <div className="dashboard-shell">
      <header className="dashboard-header">
        <div>
          <p className="application-name">Agente CPI</p>

          <h1>Central de automação</h1>

          <p className="header-description">
            Controle operacional das pesquisas CPI Central e DeskData.
          </p>
        </div>

        <StatusIndicator
          active={agent.safeMode}
          label={agent.safeMode ? 'Modo seguro ativo' : 'Modo seguro inativo'}
        />
      </header>

      <main className="dashboard-content">
        <section className="status-grid" aria-label="Estados operacionais">
          <article className="status-panel">
            <div className="status-panel-heading">
              <div>
                <span className="panel-label">Navegador</span>

                <strong data-testid="browser-state">{formatState(browser.state)}</strong>
              </div>

              <StatusIndicator
                active={browserRunning}
                label={browserRunning ? 'Conectado' : 'Fechado'}
              />
            </div>

            <p>
              Perfil persistente:
              <span className="path-value">
                {browser.profilePath || 'Aguardando inicialização'}
              </span>
            </p>

            <div className="button-row">
              <button
                className="button primary"
                type="button"
                disabled={operationBusy || browser.state !== 'STOPPED'}
                onClick={() =>
                  executeAction(
                    'browser-start',

                    setBrowser,
                    'Navegador aberto com sucesso.'
                  )
                }
              >
                Abrir navegador
              </button>

              <button
                className="button secondary"
                type="button"
                disabled={operationBusy || !browserRunning}
                onClick={() =>
                  executeAction(
                    'browser-stop',
                    window.agent.browser.stop,
                    setBrowser,
                    'Navegador encerrado com segurança.'
                  )
                }
              >
                Fechar navegador
              </button>
            </div>
          </article>

          <article className="status-panel">
            <div className="status-panel-heading">
              <div>
                <span className="panel-label">Agente</span>

                <strong data-testid="agent-state">{formatState(agent.state)}</strong>
              </div>

              <StatusIndicator
                active={!agentStopped}

              </div>

            <div className="button-row agent-controls">
