import type { BrowserContext, Page } from 'playwright'
import type { CpiQueueMonitorOptions, CpiQueueMonitorResult } from '../cpi/cpi-queue-monitor'
import type { BrowserManagerSnapshot } from '../../shared/types/browser-api'
import type { PlatformSessionsSnapshot } from '../../shared/types/session-api'
import type { AgentControlSnapshot } from '../../shared/types/agent-control-api'
import { AgentController } from './agent-controller'

export interface AgentRuntimeBrowser {
  getSnapshot(): BrowserManagerSnapshot
  getContext(): BrowserContext
}

export interface AgentRuntimeSessionService {
  getSnapshot(): Promise<PlatformSessionsSnapshot>
}

export interface AgentRuntimeFilter {
  applyCriminalForumFilter(page: Page): Promise<void>
}

export interface AgentRuntimeMonitor {
  waitForResearch(page: Page, options?: CpiQueueMonitorOptions): Promise<CpiQueueMonitorResult>
}

export class AgentRuntimeService {
  private abortController: AbortController | null = null
  private monitorPromise: Promise<void> | null = null
  private cpiPage: Page | null = null

  constructor(
    private readonly controller: AgentController,
    private readonly browser: AgentRuntimeBrowser,
    private readonly sessions: AgentRuntimeSessionService,
    private readonly filter: AgentRuntimeFilter,
    private readonly monitor: AgentRuntimeMonitor
  ) {}

  getSnapshot(): AgentControlSnapshot {
    return this.controller.getSnapshot()
  }

  async start(): Promise<AgentControlSnapshot> {
    if (this.controller.getState() !== 'PARADO') {
      throw new Error(`O agente não pode iniciar no estado ${this.controller.getState()}`)
    }

    if (this.browser.getSnapshot().state !== 'RUNNING') {
      throw new Error('O navegador precisa estar aberto')
    }

    const sessions = await this.sessions.getSnapshot()

    if (!sessions.allAuthenticated) {
      throw new Error('CPI Central e DeskData precisam estar autenticadas')
    }

    const cpiPage = this.findCpiQueuePage()

    this.controller.start()

    try {
      await this.filter.applyCriminalForumFilter(cpiPage)

      this.controller.markReady()
      this.controller.beginWaitingForResearch()

      this.cpiPage = cpiPage
      this.startQueueMonitor(cpiPage)

      return this.getSnapshot()
    } catch (error) {
      this.cpiPage = null
      this.controller.stopSafely()

      throw error
    }
  }

  async pause(): Promise<AgentControlSnapshot> {
    if (this.controller.getState() === 'PARADO') {
      throw new Error('O agente já está parado')
    }

    this.controller.requestPause()
    await this.abortCurrentMonitor()

    return this.getSnapshot()
  }

  async resume(): Promise<AgentControlSnapshot> {
    const resumedState = this.controller.resume()

    if (resumedState === 'AGUARDANDO_NOVA_PESQUISA' && this.cpiPage) {
      this.startQueueMonitor(this.cpiPage)
    }

    return this.getSnapshot()
  }

  async stop(): Promise<AgentControlSnapshot> {
    await this.abortCurrentMonitor()

    this.cpiPage = null
    this.controller.stopSafely()

    return this.getSnapshot()
  }

  private findCpiQueuePage(): Page {
    const page = this.browser
      .getContext()
      .pages()
      .find((candidate) => {
        try {
          const url = new URL(candidate.url())

          return (
            url.hostname === 'adm.cpicentral.com.br' &&
            url.pathname
              .toLowerCase()
              .endsWith('/zpersespecial/pesquisas/aguardando/mod_t_aguardando.aspx')
          )
        } catch {
          return false
        }
      })

    if (!page) {
      throw new Error('A página autenticada da fila da CPI não foi localizada')
    }

    return page
  }

  private startQueueMonitor(page: Page): void {
    if (this.monitorPromise) {
      throw new Error('O monitor da fila já está em execução')
    }

    const abortController = new AbortController()

    this.abortController = abortController

    const monitorPromise = this.monitor
      .waitForResearch(page, {
        signal: abortController.signal,

        isPauseRequested: () =>
          this.controller.getState() === 'PAUSADO' || this.controller.getState() === 'PAUSANDO'
      })
      .then((result) => {
        if (
          result.status === 'RESEARCH_AVAILABLE' &&
          !abortController.signal.aborted &&
          this.controller.getState() === 'AGUARDANDO_NOVA_PESQUISA'
        ) {
          this.controller.beginProcessing()
        }
      })
      .catch((error: unknown) => {
        if (abortController.signal.aborted) {
          return
        }

        if (this.controller.getState() === 'AGUARDANDO_NOVA_PESQUISA') {
          this.controller.markTemporaryError()
          this.controller.recordFailure()
        }

        console.error('Falha no monitoramento da fila da CPI:', error)
      })
      .finally(() => {
        if (this.abortController === abortController) {
          this.abortController = null
        }

        if (this.monitorPromise === monitorPromise) {
          this.monitorPromise = null
        }
      })

    this.monitorPromise = monitorPromise
  }

  private async abortCurrentMonitor(): Promise<void> {
    const currentPromise = this.monitorPromise

    this.abortController?.abort()

    if (currentPromise) {
      await currentPromise
    }

    this.abortController = null
    this.monitorPromise = null
  }
}

import type { AgentState } from '../../domain/agent/agent-state'
import { AgentStateMachine } from '../../domain/agent/agent-state-machine'
import { parseAppConfig, type AppConfig } from '../../infrastructure/config/app-config'
import type { AgentControlSnapshot } from '../../shared/types/agent-control-api'
import { CircuitBreaker } from './circuit-breaker'
import { PauseController } from './pause-controller'

export type AgentSnapshot = AgentControlSnapshot

export class AgentController {
  private readonly config: AppConfig
  private readonly stateMachine: AgentStateMachine
  private readonly pauseController: PauseController
  private readonly circuitBreaker: CircuitBreaker
  private resumeState: AgentState = 'PRONTO'

  constructor(configInput: unknown = {}) {
    this.config = parseAppConfig(configInput)
    this.stateMachine = new AgentStateMachine()
    this.pauseController = new PauseController()
    this.circuitBreaker = new CircuitBreaker(this.config.maxConsecutiveErrors)
  }

  getState(): AgentState {
    return this.stateMachine.getState()
  }

  getSnapshot(): AgentSnapshot {
    return {
      state: this.getState(),
      pauseRequested: this.pauseController.isPauseRequested(),
      consecutiveErrors: this.circuitBreaker.getFailureCount(),
      circuitOpen: this.circuitBreaker.isOpen(),
      safeMode: this.config.safeMode
    }
  }

  start(): AgentState {
    return this.stateMachine.transitionTo('INICIALIZANDO')
  }

  markWaitingForLogin(): AgentState {
    return this.stateMachine.transitionTo('AGUARDANDO_LOGIN')
  }

  markReady(): AgentState {
    return this.stateMachine.transitionTo('PRONTO')
  }

  beginWaitingForResearch(): AgentState {
    return this.stateMachine.transitionTo('AGUARDANDO_NOVA_PESQUISA')
  }

  beginProcessing(): AgentState {
    return this.stateMachine.transitionTo('PROCESSANDO')
  }

  markTemporaryError(): AgentState {
    return this.stateMachine.transitionTo('ERRO_TEMPORARIO')
  }

  requestPause(): AgentState {
    this.resumeState = this.getState()
    this.pauseController.requestPause()
    this.stateMachine.transitionTo('PAUSANDO')

    return this.checkpoint()
  }

  checkpoint(): AgentState {
    if (this.getState() === 'PAUSANDO' && this.pauseController.shouldPauseAtCheckpoint()) {
      return this.stateMachine.transitionTo('PAUSADO')
    }

    return this.getState()
  }

  enterAtomicSection(): void {
    this.pauseController.enterAtomicSection()
  }

  leaveAtomicSection(): AgentState {
    this.pauseController.leaveAtomicSection()

    return this.checkpoint()
  }

  resume(): AgentState {
    if (this.getState() !== 'PAUSADO') {
      throw new Error(`Não é possível continuar o agente no estado ${this.getState()}`)
    }

    this.pauseController.resume()

    const targetState = this.stateMachine.canTransitionTo(this.resumeState)
      ? this.resumeState
      : 'PRONTO'

    return this.stateMachine.transitionTo(targetState)
  }

  stopSafely(): AgentState {
    if (this.getState() === 'PARADO') {
      return 'PARADO'
    }

    if (this.getState() !== 'ENCERRANDO') {
      this.stateMachine.transitionTo('ENCERRANDO')
    }

    this.pauseController.resume()

    return this.stateMachine.transitionTo('PARADO')
  }

  recordFailure(): AgentSnapshot {
    this.circuitBreaker.recordFailure()

    if (
      this.circuitBreaker.isOpen() &&
      this.getState() !== 'PAUSADO_POR_SEGURANCA' &&
      this.stateMachine.canTransitionTo('PAUSADO_POR_SEGURANCA')
    ) {
      this.stateMachine.transitionTo('PAUSADO_POR_SEGURANCA')
    }

    return this.getSnapshot()
  }

  recordSuccess(): AgentSnapshot {
    this.circuitBreaker.recordSuccess()

    return this.getSnapshot()
  }
}

export class PauseController {
  private pauseRequested = false
  private atomicDepth = 0

  requestPause(): void {
    this.pauseRequested = true
  }

  resume(): void {
    this.pauseRequested = false
  }

  enterAtomicSection(): void {
    this.atomicDepth += 1
  }

  leaveAtomicSection(): void {
    this.atomicDepth = Math.max(0, this.atomicDepth - 1)
  }

  isPauseRequested(): boolean {
    return this.pauseRequested
  }

  shouldPauseAtCheckpoint(): boolean {
    return this.pauseRequested && this.atomicDepth === 0
  }
}

import type { AgentState } from './agent-state'

export const ALLOWED_AGENT_TRANSITIONS: Readonly<Record<AgentState, readonly AgentState[]>> = {
  PARADO: ['INICIALIZANDO'],

  INICIALIZANDO: ['AGUARDANDO_LOGIN', 'PRONTO', 'PAUSADO_POR_SEGURANCA', 'ENCERRANDO'],

  AGUARDANDO_LOGIN: ['PRONTO', 'PAUSANDO', 'PAUSADO_POR_SEGURANCA', 'ENCERRANDO'],

  PRONTO: [
    'PROCESSANDO',
    'AGUARDANDO_NOVA_PESQUISA',
    'PAUSANDO',
    'PAUSADO_POR_SEGURANCA',
    'ENCERRANDO'
  ],

  PROCESSANDO: [
    'AGUARDANDO_NOVA_PESQUISA',
    'PAUSANDO',
    'ERRO_TEMPORARIO',
    'PAUSADO_POR_SEGURANCA',
    'ENCERRANDO'
  ],

  PAUSANDO: ['PAUSADO', 'PAUSADO_POR_SEGURANCA', 'ENCERRANDO'],

  PAUSADO: ['AGUARDANDO_LOGIN', 'PRONTO', 'PROCESSANDO', 'AGUARDANDO_NOVA_PESQUISA', 'ENCERRANDO'],

  AGUARDANDO_NOVA_PESQUISA: [
    'PROCESSANDO',
    'PAUSANDO',
    'ERRO_TEMPORARIO',
    'PAUSADO_POR_SEGURANCA',
    'ENCERRANDO'
  ],

  ERRO_TEMPORARIO: [
    'PROCESSANDO',
    'AGUARDANDO_NOVA_PESQUISA',
    'PAUSANDO',
    'PAUSADO_POR_SEGURANCA',
    'ENCERRANDO'
  ],

  PAUSADO_POR_SEGURANCA: ['AGUARDANDO_LOGIN', 'PRONTO', 'PAUSADO', 'ENCERRANDO'],

  ENCERRANDO: ['PARADO']
}

export class AgentStateMachine {
  private currentState: AgentState

  constructor(initialState: AgentState = 'PARADO') {
    this.currentState = initialState
  }

  getState(): AgentState {
    return this.currentState
  }

  canTransitionTo(nextState: AgentState): boolean {
    return ALLOWED_AGENT_TRANSITIONS[this.currentState].includes(nextState)
  }

  transitionTo(nextState: AgentState): AgentState {
    if (!this.canTransitionTo(nextState)) {
      throw new Error(`Transição inválida do agente: ${this.currentState} -> ${nextState}`)
    }

    this.currentState = nextState

    return this.currentState
  }
}

import type { BrowserContext } from 'playwright'
import type { BrowserLauncher, BrowserManagerSnapshot, BrowserManagerState } from './browser-types'

export class BrowserManager {
  private state: BrowserManagerState = 'STOPPED'
  private context: BrowserContext | null = null
  private startedAt: Date | null = null

  constructor(
    private readonly launcher: BrowserLauncher,
    private readonly profilePath: string
  ) {}

  getSnapshot(): BrowserManagerSnapshot {
    return {
      state: this.state,
      profilePath: this.profilePath,
      startedAt: this.startedAt?.toISOString() ?? null
    }
  }

  getContext(): BrowserContext {
    if (this.state !== 'RUNNING' || !this.context) {
      throw new Error('Contexto do navegador indisponível')
    }

    return this.context
  }

  async start(): Promise<BrowserManagerSnapshot> {
    if (this.state !== 'STOPPED') {
      throw new Error(`Navegador não pode iniciar no estado ${this.state}`)
    }

    this.state = 'STARTING'

    try {
      this.context = await this.launcher.launchPersistentContext(this.profilePath)

      this.startedAt = new Date()
      this.state = 'RUNNING'

      return this.getSnapshot()
    } catch (error) {
      this.context = null
      this.startedAt = null
      this.state = 'STOPPED'

      throw error
    }
  }

  async stop(): Promise<BrowserManagerSnapshot> {
    if (this.state === 'STOPPED') {
      return this.getSnapshot()
    }

    if (this.state !== 'RUNNING') {
      throw new Error(`Navegador não pode parar no estado ${this.state}`)
    }

    this.state = 'STOPPING'

    const context = this.context

    try {
      await context?.close()
    } finally {
      this.context = null
      this.startedAt = null
      this.state = 'STOPPED'
    }

    return this.getSnapshot()
  }
}

import { parsePlatformUrls, type PlatformUrls } from './platform-urls'

export interface ManagedPage {
  url(): string
  isClosed(): boolean
  goto(
    url: string,
    options?: {
      waitUntil?: 'domcontentloaded'
      timeout?: number
    }
  ): Promise<unknown>
  bringToFront(): Promise<void>
}

export interface ManagedBrowserContext {
  pages(): ManagedPage[]
  newPage(): Promise<ManagedPage>
}

export interface PlatformPagesSnapshot {
  cpiPageUrl: string
  deskDataPageUrl: string
}

export class PlatformPageManager {
  private readonly urls: PlatformUrls

  constructor(urlInput: unknown = undefined) {
    this.urls = urlInput === undefined ? parsePlatformUrls() : parsePlatformUrls(urlInput)
  }

  async ensurePages(context: ManagedBrowserContext): Promise<PlatformPagesSnapshot> {
    const usedPages = new Set<ManagedPage>()

    const cpiPage = await this.resolvePage(context, this.urls.cpiQueueUrl, usedPages)

    usedPages.add(cpiPage)

    const deskDataPage = await this.resolvePage(context, this.urls.deskDataPeopleUrl, usedPages)

    await cpiPage.bringToFront()

    return {
      cpiPageUrl: cpiPage.url(),
      deskDataPageUrl: deskDataPage.url()
    }
  }

  private async resolvePage(
    context: ManagedBrowserContext,
    targetUrl: string,
    usedPages: ReadonlySet<ManagedPage>
  ): Promise<ManagedPage> {
    const availablePages = context
      .pages()
      .filter((page) => !page.isClosed() && !usedPages.has(page))

    const matchingPage = availablePages.find((page) => this.hasSameHostname(page.url(), targetUrl))

    const blankPage = availablePages.find((page) => page.url() === 'about:blank')

    const page = matchingPage ?? blankPage ?? (await context.newPage())

    if (page.url() !== targetUrl) {
      await page.goto(targetUrl, {
        waitUntil: 'domcontentloaded',
        timeout: 45_000
      })
    }

    return page
  }

  private hasSameHostname(currentUrl: string, targetUrl: string): boolean {
    try {
      return new URL(currentUrl).hostname === new URL(targetUrl).hostname
    } catch {
      return false
    }
  }
}

    const deskDataEvidence = await this.inspectPage(deskDataPage)

      return new URL(pageUrl).hostname === hostname

import type { PlatformSessionState } from '../../shared/types/session-api'

export interface PageSessionEvidence {
  url: string
  hasVisiblePasswordInput: boolean
}

export function detectCpiSession(evidence: PageSessionEvidence | null): PlatformSessionState {
  if (!evidence) {
    return 'INDETERMINATE'
  }

  if (evidence.hasVisiblePasswordInput) {
    return 'WAITING_LOGIN'
  }

  const url = new URL(evidence.url)

  if (url.pathname.toLowerCase().endsWith('/access_fault.aspx')) {
    return 'WAITING_LOGIN'
  }

  if (
    url.hostname === 'adm.cpicentral.com.br' &&
    url.pathname.toLowerCase().endsWith('/zpersespecial/pesquisas/aguardando/mod_t_aguardando.aspx')
  ) {
    return 'AUTHENTICATED'
  }

  return 'INDETERMINATE'
}

export function detectDeskDataSession(evidence: PageSessionEvidence | null): PlatformSessionState {
  if (!evidence) {
    return 'INDETERMINATE'
  }

  if (evidence.hasVisiblePasswordInput) {
    return 'WAITING_LOGIN'
  }

  const url = new URL(evidence.url)
  const pathname = url.pathname.toLowerCase()

  if (
    url.hostname === 'deskdata.com.br' &&
    (pathname === '/pessoas' || pathname.startsWith('/pessoas/'))
  ) {
    return 'AUTHENTICATED'
  }

  if (pathname.includes('/login') || pathname.includes('/entrar') || pathname.includes('/acesso')) {
    return 'WAITING_LOGIN'
  }

  return 'INDETERMINATE'
}

import { mkdir } from 'node:fs/promises'
import { join, resolve } from 'node:path'

export interface RuntimePaths {
  rootDir: string
  browserProfileDir: string
  downloadsDir: string
  screenshotsDir: string
  logsDir: string
  databaseDir: string
  databaseFile: string
  configDir: string
}

export function resolveRuntimePaths(rootPath: string): RuntimePaths {
  if (!rootPath.trim()) {
    throw new Error('Diretório operacional raiz não informado')
  }

  const rootDir = resolve(rootPath)
  const databaseDir = join(rootDir, 'database')

  return {
    rootDir,
    browserProfileDir: join(rootDir, 'browser-profile'),
    downloadsDir: join(rootDir, 'downloads'),
    screenshotsDir: join(rootDir, 'screenshots'),
    logsDir: join(rootDir, 'logs'),
    databaseDir,
    databaseFile: join(databaseDir, 'agent.db'),
    configDir: join(rootDir, 'config')
  }
}

export async function ensureRuntimeDirectories(paths: RuntimePaths): Promise<void> {
  const directories = [
    paths.rootDir,
    paths.browserProfileDir,
    paths.downloadsDir,
    paths.screenshotsDir,
    paths.logsDir,
    paths.databaseDir,
    paths.configDir
  ]

  await Promise.all(
    directories.map((directory) =>
      mkdir(directory, {
        recursive: true
      })
    )
  )
}

import type { BrowserContext, Page } from 'playwright'
import type { BrowserManagerSnapshot } from '@/shared/types/browser-api'
import type { PlatformSessionsSnapshot } from '@/shared/types/session-api'
import { describe, expect, it } from 'vitest'
import {
  AgentRuntimeService,

  type AgentRuntimeFilter,
  type AgentRuntimeMonitor,
  type AgentRuntimeSessionService
} from '@/automation/workflow/agent-runtime-service'
import { AgentController } from '@/automation/workflow/agent-controller'
import type { CpiQueueMonitorOptions } from '@/automation/cpi/cpi-queue-monitor'

function createCpiPage(): Page {
  return {
    url: () =>
      'https://adm.cpicentral.com.br/zpersEspecial/pesquisas/aguardando/mod_t_aguardando.aspx'
  } as unknown as Page
}

class FakeBrowser implements AgentRuntimeBrowser {
  constructor(
    private readonly page: Page,
    private readonly running = true
  ) {}

  getSnapshot(): BrowserManagerSnapshot {
    return {
      state: this.running ? ('RUNNING' as const) : ('STOPPED' as const),
      profilePath: 'C:\\AgenteCPI\\browser-profile',
      startedAt: this.running ? '2026-07-14T08:00:00.000Z' : null
    }
  }

  getContext(): BrowserContext {
    return {
      pages: () => [this.page]
    } as unknown as BrowserContext
  }
}

class FakeSessions implements AgentRuntimeSessionService {
  constructor(private readonly authenticated = true) {}

  async getSnapshot(): Promise<PlatformSessionsSnapshot> {
    return {
      cpi: {
        state: this.authenticated ? ('AUTHENTICATED' as const) : ('WAITING_LOGIN' as const),
        url: null
      },
      deskData: {
        state: 'AUTHENTICATED' as const,
        url: null
      },
      allAuthenticated: this.authenticated,
      checkedAt: '2026-07-14T08:00:00.000Z'
    }
  }
}

class FakeFilter implements AgentRuntimeFilter {
  public calls = 0

  async applyCriminalForumFilter(): Promise<void> {
    this.calls += 1
  }
}

class ControlledMonitor implements AgentRuntimeMonitor {
  public calls = 0
  public lastOptions: CpiQueueMonitorOptions | null = null

  private resolver:
    ((result: { status: 'RESEARCH_AVAILABLE' | 'PAUSED'; checks: number }) => void) | null = null

  waitForResearch(
    _page: Page,
    options: CpiQueueMonitorOptions
  ): Promise<{
    status: 'RESEARCH_AVAILABLE' | 'PAUSED'
    checks: number
  }> {
    this.calls += 1
    this.lastOptions = options

    return new Promise((resolve) => {
      this.resolver = resolve

      options.signal?.addEventListener(
        'abort',
        () => {
          resolve({
            status: 'PAUSED',
            checks: 1
          })
        },
        {
          once: true
        }
      )
    })
  }

  releaseResearch(): void {
    this.resolver?.({
      status: 'RESEARCH_AVAILABLE',
      checks: 1
    })
  }
}

describe('AgentRuntimeService', () => {
  it('deve aplicar o filtro e iniciar o monitor da fila', async () => {
    const controller = new AgentController()
    const filter = new FakeFilter()
    const monitor = new ControlledMonitor()

    const service = new AgentRuntimeService(
      controller,
      new FakeBrowser(createCpiPage()),
      new FakeSessions(),
      filter,
      monitor
    )

    const snapshot = await service.start()

    expect(snapshot.state).toBe('AGUARDANDO_NOVA_PESQUISA')
    expect(filter.calls).toBe(1)
    expect(monitor.calls).toBe(1)
  })

  it('deve mudar para PROCESSANDO quando uma pesquisa surgir', async () => {
    const controller = new AgentController()
    const monitor = new ControlledMonitor()

    const service = new AgentRuntimeService(
      controller,
      new FakeBrowser(createCpiPage()),
      new FakeSessions(),
      new FakeFilter(),
      monitor
    )

    await service.start()

    monitor.releaseResearch()

    await new Promise((resolve) => setTimeout(resolve, 0))

    expect(service.getSnapshot().state).toBe('PROCESSANDO')
  })

  it('deve pausar o monitor de forma segura', async () => {
    const controller = new AgentController()
    const monitor = new ControlledMonitor()

    const service = new AgentRuntimeService(
      controller,
      new FakeBrowser(createCpiPage()),
      new FakeSessions(),
      new FakeFilter(),
      monitor
    )

    await service.start()

    const snapshot = await service.pause()

    expect(snapshot.state).toBe('PAUSADO')
    expect(monitor.lastOptions?.signal?.aborted).toBe(true)
  })

  it('deve reiniciar o monitor ao continuar', async () => {
    const controller = new AgentController()
    const monitor = new ControlledMonitor()

    const service = new AgentRuntimeService(
      controller,
      new FakeBrowser(createCpiPage()),
      new FakeSessions(),
      new FakeFilter(),
      monitor
    )

    await service.start()
    await service.pause()
    await service.resume()

    expect(service.getSnapshot().state).toBe('AGUARDANDO_NOVA_PESQUISA')

    expect(monitor.calls).toBe(2)
  })

  it('deve impedir o início sem autenticação', async () => {
    const service = new AgentRuntimeService(
      new AgentController(),
      new FakeBrowser(createCpiPage()),
      new FakeSessions(false),
      new FakeFilter(),
      new ControlledMonitor()
    )

      'CPI Central e DeskData precisam estar autenticadas'
    )

    expect(service.getSnapshot().state).toBe('PARADO')
  })

  it('deve impedir o início com o navegador fechado', async () => {
